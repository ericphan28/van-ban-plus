  phuongThuc?: "GET" | "POST" | "PATCH" | "DELETE";
  than?: unknown;

  hanMs?: number;
  /** Người dùng hủy (nút «Hủy», rời trang). */
  signal?: AbortSignal;
  dau?: Record<string, string>;
};

const HAN_YEU_CAU_MS = 15_000;
const MA_CLOUDFLARE_HET_GIO = new Set([504, 522, 524]);

/** Ghép hạn chờ + tín hiệu hủy của người dùng; `don()` gỡ hẹn giờ khi xong. Không có người hủy → giữ đúng hành vi cũ
 *  (`AbortSignal.timeout`, hoặc không signal khi trình duyệt chưa có) để bài kiểm cũ xanh. Không dùng `AbortSignal.any`
 *  (Safari < 17.4 chưa có). */
function tinHieu(hanMs: number, nguoiHuy?: AbortSignal): { signal?: AbortSignal; don: () => void } {
  if (!nguoiHuy) {
    const coHan = typeof AbortSignal !== "undefined" && typeof AbortSignal.timeout === "function";
    return { signal: coHan ? AbortSignal.timeout(hanMs) : undefined, don: () => {} };
  }
  const c = new AbortController();
  const hen = setTimeout(() => c.abort(), hanMs);
  const huy = () => c.abort();
  if (nguoiHuy.aborted) huy();
  else nguoiHuy.addEventListener("abort", huy, { once: true });
  return { signal: c.signal, don: () => { clearTimeout(hen); nguoiHuy.removeEventListener("abort", huy); } };
}

async function guiTho(duong: string, t: TuyChonGoi, bearer: string | null) {
  const dau: Record<string, string> = { ...(t.dau ?? {}) };
  if (t.than !== undefined) dau["Content-Type"] = "application/json";
  if (bearer) dau.Authorization = `Bearer ${bearer}`;
  const th = tinHieu(t.hanMs ?? HAN_YEU_CAU_MS, t.signal);
  let r: Response;
  try {
    r = await ch.fetch(ch.goc + duong, {
      method: t.phuongThuc ?? "GET", headers: dau, credentials: "include",
      body: t.than === undefined ? undefined : JSON.stringify(t.than),
      signal: th.signal,
    });
  } catch {
    th.don();
    if (t.signal?.aborted) throw new LoiApi(0, "DA_HUY", "Đã hủy.");
    throw new LoiApi(0, "MAT_MANG", "Không kết nối được máy chủ. Kiểm tra mạng rồi thử lại.");
  }
  th.don();
  const j = await docThan(r);
  if (MA_CLOUDFLARE_HET_GIO.has(r.status) && j.khongDoc) {
    if (duong.startsWith("/api/v2/tac-vu/")) {
      throw new LoiApi(r.status, "QUA_GIO_CHO",

const HAN_TAI_LEN_MS = 10 * 60_000;

/** URL ký (tải lên / tải về) có đúng nguồn Storage cấu hình không. Dùng trước mọi PUT, `iframe`, `img`, liên kết tải về. */
export function laUrlStorage(url: string): boolean {
  try { return new URL(url).origin === ch.nguonStorage; } catch { return false; }
}

/** XHR để có tiến trình tải lên (fetch chưa báo được tiến trình gửi). */
function putXhr(url: string, tep: Blob, mime: string, signal: AbortSignal | undefined, khiTien: (da: number, tong: number) => void) {
  return new Promise<void>((ok, hong) => {
    const x = ch.taoXhr!();
    const huy = () => x.abort();
    x.open("PUT", url);
    x.setRequestHeader("Content-Type", mime);
    x.timeout = HAN_TAI_LEN_MS;
    x.upload.onprogress = (e) => { if (e.lengthComputable) khiTien(e.loaded, e.total); };
    x.onload = () => { signal?.removeEventListener("abort", huy);
      x.status >= 200 && x.status < 300 ? ok() : hong(new LoiApi(x.status, "TAI_LEN_LOI", "Kho tệp từ chối tệp này. Thử lại sau.")); };
    x.onerror = () => { signal?.removeEventListener("abort", huy); hong(new LoiApi(0, "MAT_MANG", "Mất kết nối khi tải tệp lên. Thử lại.")); };
    x.ontimeout = x.onerror;
    x.onabort = () => hong(new LoiApi(0, "DA_HUY", "Đã hủy tải lên."));
    if (signal?.aborted) { hong(new LoiApi(0, "DA_HUY", "Đã hủy tải lên.")); return; }
    signal?.addEventListener("abort", huy, { once: true });
    x.send(tep);
  });
}

/** PUT tệp lên đường ký của Storage (khác nguồn API). Không Bearer, không cookie. Chỉ nhận đúng nguồn Storage cấu hình.
 *  Có `khiTien` và môi trường có XHR → dùng XHR để báo tiến trình; không thì fetch. */
export async function taiLenStorage(url: string, tep: Blob, mime: string, signal?: AbortSignal,
  khiTien?: (da: number, tong: number) => void): Promise<void> {
  if (!laUrlStorage(url)) throw new LoiApi(0, "TEP_KHONG_HOP_LE", "Đường tải lên không hợp lệ.");
  if (khiTien && ch.taoXhr) return putXhr(url, tep, mime, signal, khiTien);
  const th = tinHieu(HAN_TAI_LEN_MS, signal);
  let r: Response;
  try {
    r = await ch.fetch(url, { method: "PUT", headers: { "Content-Type": mime }, body: tep, credentials: "omit", signal: th.signal });
  } catch {
    th.don();
    if (signal?.aborted) throw new LoiApi(0, "DA_HUY", "Đã hủy tải lên.");
    throw new LoiApi(0, "MAT_MANG", "Mất kết nối khi tải tệp lên. Thử lại.");
  }
  th.don();
  if (!r.ok) throw new LoiApi(r.status, "TAI_LEN_LOI", "Kho tệp từ chối tệp này. Thử lại sau.");
}

export function ddMMSangIso(s: string | null | undefined): string | null {
  const m = /^(\d{1,2})\/(\d{1,2})\/(\d{4})$/.exec((s ?? "").trim());

  const iso = `${m[3]}-${m[2].padStart(2, "0")}-${m[1].padStart(2, "0")}`;

export function dongNgayNd30(diaDanh: string, iso: string | null | undefined): string {
  const dd = diaDanh.trim() || "......";

  const ngay = d < 10 ? `0${d}` : String(d);
  const thang = m <= 2 ? `0${m}` : String(m);
  return `${dd}, ngày ${ngay} tháng ${thang} năm ${y}`;
}

import type { Huong, VanBan } from "@/api/kho";

/** Trạng thái biểu mẫu: mọi ô là chuỗi (mảng = mỗi dòng một mục; thẻ = phân cách dấu phẩy). */
export type BanNhap = {
  huong: Huong; thu_muc_id: string | null; so: string; ky_hieu: string; ngay_ban_hanh: string; co_quan: string; loai: string;
  trich_yeu: string; do_khan: string; do_mat: string; so_den: string; ngay_den: string; han_xu_ly: string; giao_cho: string;

};
const CHUOI = ["so", "ky_hieu", "co_quan", "loai", "trich_yeu", "giao_cho", "nguoi_ky", "chuc_danh", "noi_dung", "tom_tat", "ghi_chu"] as const;
const NGAY = ["ngay_ban_hanh", "ngay_den", "han_xu_ly"] as const;
const DONG = ["can_cu", "noi_nhan"] as const;

export const banNhapTrong = (huong: Huong): BanNhap => ({
  huong, thu_muc_id: null, so: "", ky_hieu: "", ngay_ban_hanh: "", co_quan: "", loai: "", trich_yeu: "", do_khan: "thuong",
  do_mat: "thuong", so_den: "", ngay_den: "", han_xu_ly: "", giao_cho: "", nguoi_ky: "", chuc_danh: "", can_cu: "",
  noi_nhan: "", the: "", noi_dung: "", tom_tat: "", ghi_chu: "",

export function tuVanBan(v: VanBan): BanNhap {
  return {
    huong: v.huong, thu_muc_id: v.thu_muc_id, so: v.so ?? "", ky_hieu: v.ky_hieu ?? "", ngay_ban_hanh: v.ngay_ban_hanh ?? "",
    co_quan: v.co_quan ?? "", loai: v.loai ?? "", trich_yeu: v.trich_yeu, do_khan: v.do_khan || "thuong", do_mat: v.do_mat || "thuong",
    so_den: v.so_den == null ? "" : String(v.so_den), ngay_den: v.ngay_den ?? "", han_xu_ly: v.han_xu_ly ?? "",
    giao_cho: v.giao_cho ?? "", nguoi_ky: v.nguoi_ky ?? "", chuc_danh: v.chuc_danh ?? "", can_cu: v.can_cu.join("\n"),
    noi_nhan: v.noi_nhan.join("\n"), the: v.the.join(", "), noi_dung: v.noi_dung ?? "", tom_tat: v.tom_tat ?? "", ghi_chu: v.ghi_chu ?? "",
  };
}

const rongThanhNull = (s: string) => (s.trim() === "" ? null : s.trim());
const dong = (s: string) => s.split(/\r?\n/).map((x) => x.trim()).filter(Boolean);

/** Thân đầy đủ cho POST /kho/van-ban (API kiểm lại tất cả). */
export function sangThan(b: BanNhap): Record<string, unknown> {
  const t: Record<string, unknown> = { huong: b.huong, thu_muc_id: b.thu_muc_id, do_khan: b.do_khan, do_mat: b.do_mat };
  for (const k of CHUOI) t[k] = rongThanhNull(b[k]);
  for (const k of NGAY) t[k] = rongThanhNull(b[k]);
  for (const k of DONG) t[k] = dong(b[k]);
  t.the = b.the.split(",").map((x) => x.trim()).filter(Boolean);
  t.so_den = b.so_den.trim() === "" ? null : Number(b.so_den.trim());
  return t;
}

/** Chỉ trường đã đổi (cho PATCH). So sánh trên thân đã chuẩn hóa để «a, b» và «a,b» không tính là đổi. */
export function khacBiet(goc: BanNhap, moi: BanNhap): Record<string, unknown> {
  const a = sangThan(goc); const b = sangThan(moi);
  const kq: Record<string, unknown> = {};
  for (const k of Object.keys(b)) if (JSON.stringify(a[k]) !== JSON.stringify(b[k])) kq[k] = b[k];
  return kq;
}

export function kiemBanNhap(b: BanNhap): string | null {
  if (!b.trich_yeu.trim()) return "Thiếu trích yếu.";
  if (b.so_den.trim() !== "" && !/^\d+$/.test(b.so_den.trim())) return "Số đến phải là số nguyên.";
  return null;
}

const HUONG_TX: Record<string, Huong> = { Den: "den", Di: "di", NoiBo: "noi_bo" };
const KHAN_TX: Record<string, string> = { Thuong: "thuong", Khan: "khan", ThuongKhan: "thuong_khan", HoaToc: "hoa_toc" };
const MAT_TX: Record<string, string> = { Thuong: "thuong", Mat: "mat", ToiMat: "toi_mat", TuyetMat: "tuyet_mat" };

/** Kết quả trich-xuat (cau-truc) → biểu mẫu. Quét mặc định là văn bản đến, ngày đến = hôm nay. */
export function tuTrichXuat(k: TrichXuat, homNay: string): BanNhap {
  const huong = HUONG_TX[k.huong_van_ban ?? ""] ?? "den";
  const { so, ky_hieu } = tachSoKyHieu(k.so_van_ban);
  return {
    ...banNhapTrong(huong), so, ky_hieu, ngay_ban_hanh: ddMMSangIso(k.ngay_ban_hanh) ?? "",
    co_quan: k.co_quan_ban_hanh ?? "", loai: laMaLoai(k.loai_van_ban) ? k.loai_van_ban : "", trich_yeu: k.trich_yeu ?? "",
    do_khan: KHAN_TX[k.do_khan ?? ""] ?? "thuong", do_mat: MAT_TX[k.do_mat ?? ""] ?? "thuong",
    nguoi_ky: k.nguoi_ky ?? "", chuc_danh: k.chuc_danh_ky ?? "", noi_dung: k.noi_dung ?? "",
    noi_nhan: (k.noi_nhan ?? []).join("\n"), can_cu: (k.can_cu ?? []).join("\n"), ngay_den: huong === "den" ? homNay : "",
  };
}

```ts
import type { VanBan } from "@/api/kho";
import { dongNgayNd30, tenLoai } from "@/phan-he/van-ban/danh-muc";

export type BoCuc = {
  coQuanChuQuan: string; coQuan: string[]; so: string; diaDanhNgay: string; vv: string; tenLoai: string; trichYeu: string[];
  kinhGui: string[]; than: string[]; noiNhan: string[]; kyTen: string[];
};
export const boCucTrong = (): BoCuc => ({ coQuanChuQuan: "", coQuan: [], so: "", diaDanhNgay: "", vv: "", tenLoai: "", trichYeu: [],
  kinhGui: [], than: [], noiNhan: [], kyTen: [] });

const QUOC_HIEU = /^CỘNG H[OÒ]A XÃ HỘI CHỦ NGHĨA VIỆT NAM$/iu;
const TIEU_NGU = /^Độc lập\s*[-–—]\s*Tự do\s*[-–—]\s*Hạnh phúc$/iu;
const GACH = /^[-_—–=.\s]+$/;
const SO = /^Số\s*:/iu;
const NGAY = /ngày\s+\S+\s+tháng\s+\S+\s+năm\s+\S+/iu;
const VV = /^V\/v\b/iu;
const KINH_GUI = /^Kính gửi\b/iu;
const NOI_NHAN = /^Nơi nhận\s*:?/iu;
const CHO_KY = /^\((Chữ ký|Ký|Đã ký)[^)]*\)$/iu;
const DAU_DONG = /^[-+•]\s*/;
const CO_QUAN_HAI_DONG = /^(ỦY|UỶ) BAN NHÂN DÂN$|^HỘI ĐỒNG NHÂN DÂN$/u;

export const laHoa = (s: string) => /\p{L}/u.test(s) && s === s.toLocaleUpperCase("vi-VN");
const laTen = (s: string) => s.length <= 40 && /^(\p{Lu}[\p{Ll}]*)(\s\p{Lu}[\p{Ll}]*){1,5}$/u.test(s);

/** Mỗi dòng → các ô (tách cột bằng tab hoặc ≥ 3 dấu cách); bỏ markdown AI hay chèn. «Số:    /» gộp còn «Số: /». */
function chuanHoa(chu: string): string[] {
  return chu.replace(/\r\n?/g, "\n").replace(/```[a-z]*\n?/gi, "").split("\n")
    .map((d) => d.replace(/\*\*|__/g, "").replace(/^#+\s*/, "").replace(/(Số\s*:)\s+/iu, "$1 ")
      // «ngày      tháng      năm 2026» (để trống chờ điền) không được tách thành ba cột
      .replace(/(ngày)\s{2,}(tháng)\s{2,}(năm)/iu, "$1 ... $2 ... $3")
      .replace(/(ngày\s+\S+)\s{2,}(tháng\s+\S+)\s{2,}(năm)/iu, "$1 $2 $3"))
    .flatMap((d) => d.split(/\t+|\s{3,}/))
    .map((o) => o.trim())
    .filter(Boolean);
}

function laDongKy(s: string): boolean {
  return laHoa(s) && s.length <= 60 && !/[.;:]$/.test(s) && !/^(ĐIỀU|CHƯƠNG|MỤC|[IVXLC]+\.|\d+\.)/u.test(s);
}

export function phanTichVanBan(chu: string): BoCuc {
  const o = chuanHoa(chu);
  const bc = boCucTrong();
  const hoaDau: string[] = [];
  let daQuaSo = false; let trongKinhGui = false; let i = 0;
  for (; i < o.length && i < 30; i++) {
    const s = o[i];
    if (QUOC_HIEU.test(s) || TIEU_NGU.test(s) || GACH.test(s)) continue;
    if (SO.test(s) && !bc.so) { bc.so = s; daQuaSo = true; continue; }
    if (NGAY.test(s) && s.length < 120 && !bc.diaDanhNgay) { bc.diaDanhNgay = s; daQuaSo = true; continue; }
    if (VV.test(s) && !bc.vv) { bc.vv = s; continue; }
    if (KINH_GUI.test(s)) { bc.kinhGui.push(s); trongKinhGui = true; continue; }
    if (trongKinhGui && DAU_DONG.test(s)) { bc.kinhGui.push(s); continue; }
    if (!daQuaSo && laHoa(s) && s.length <= 80) { hoaDau.push(s); continue; }
    if (daQuaSo && !bc.tenLoai && !bc.vv && !trongKinhGui && laHoa(s) && s.length <= 40 && !s.endsWith(":")) { bc.tenLoai = s; continue; }
    if (bc.tenLoai && bc.trichYeu.length === 0 && !laHoa(s) && !/^Căn cứ/iu.test(s)) { bc.trichYeu.push(s); continue; }
    break;
  }
  let khongCoDau = false;
  if (!daQuaSo && !bc.vv && !bc.tenLoai) { i = 0; hoaDau.length = 0; bc.kinhGui = []; khongCoDau = true; } // toàn bộ là thân
  if (hoaDau.length >= 2 && !CO_QUAN_HAI_DONG.test(hoaDau[0])) { bc.coQuanChuQuan = hoaDau[0]; bc.coQuan = hoaDau.slice(1); }
  else bc.coQuan = hoaDau;

  // quốc hiệu, tiêu ngữ, gạch: bộ dựng Word luôn tự in → không để trong thân
  const con = khongCoDau ? o.filter((s) => !QUOC_HIEU.test(s) && !TIEU_NGU.test(s) && !GACH.test(s)) : o.slice(i);
  const iNN = con.findIndex((s) => NOI_NHAN.test(s));
  let than = iNN < 0 ? con : con.slice(0, iNN);
  const sau = iNN < 0 ? [] : con.slice(iNN + 1);
  if (iNN >= 0) { const du = con[iNN].replace(NOI_NHAN, "").trim(); if (du) sau.unshift(du); }
  for (const s of sau) {
    if (CHO_KY.test(s) || GACH.test(s)) continue;
    if (DAU_DONG.test(s)) bc.noiNhan.push(s.replace(DAU_DONG, ""));
    else bc.kyTen.push(s);
  }
  if (bc.kyTen.length === 0) {
    // Khối ký đứng trước «Nơi nhận:» (hoặc không có Nơi nhận): bóc các dòng cuối thân — họ tên, rồi các dòng chức danh in hoa.
    const ky: string[] = [];
    let j = than.length - 1;
    while (j >= 0 && CHO_KY.test(than[j])) j--;
    if (j >= 0 && laTen(than[j])) { ky.unshift(than[j]); j--; }
    while (j >= 0 && (laDongKy(than[j]) || CHO_KY.test(than[j]))) { if (!CHO_KY.test(than[j])) ky.unshift(than[j]); j--; }
    if (ky.some(laHoa)) { bc.kyTen = ky; than = than.slice(0, j + 1); }
  }
  bc.than = than.filter((s) => !CHO_KY.test(s));
  return bc;
}

const TIEN_TO_DIA = /\s(xã|phường|thị trấn|đặc khu|tỉnh|thành phố)\s+(.+)$/iu;
export function diaDanhTu(coQuan: string): string {
  const m = TIEN_TO_DIA.exec(` ${coQuan.trim()}`);
  return m ? m[2].trim() : "";
}
const TACH_CO_QUAN = /\s(XÃ|PHƯỜNG|THỊ TRẤN|ĐẶC KHU|TỈNH|THÀNH PHỐ)\s/u;
/** NĐ 30 + prompt máy chủ: tên cơ quan viết đủ («ỦY BAN NHÂN DÂN / XÃ …»), không viết tắt UBND / HĐND. */
function tachCoQuan(cq: string): string[] {
  const hoa = cq.trim().toLocaleUpperCase("vi-VN")
    .replace(/^UBND\b/u, "ỦY BAN NHÂN DÂN").replace(/^HĐND\b/u, "HỘI ĐỒNG NHÂN DÂN");
  if (!hoa) return [];
  const m = TACH_CO_QUAN.exec(hoa);
  return m && m.index > 0 ? [hoa.slice(0, m.index), hoa.slice(m.index + 1)] : [hoa];
}

export function boCucTuVanBan(vb: VanBan): BoCuc {
  const nd = vb.noi_dung ?? "";
  if (QUOC_HIEU.test(nd.split(/\r?\n/).slice(0, 12).flatMap((d) => d.split(/\t+|\s{3,}/)).map((s) => s.trim()).find((s) => QUOC_HIEU.test(s)) ?? "")) {
    return phanTichVanBan(nd);
  }
  const laCv = vb.loai === "CongVan";
  const soKh = [vb.so, vb.ky_hieu].filter((x) => x && x.trim()).join("/");
  return {
    ...boCucTrong(),
    coQuan: tachCoQuan(vb.co_quan ?? ""),
    so: `Số: ${soKh || "/"}`,
    diaDanhNgay: dongNgayNd30(diaDanhTu(vb.co_quan ?? ""), vb.ngay_ban_hanh),
    vv: laCv && vb.trich_yeu ? `V/v ${vb.trich_yeu}` : "",
    tenLoai: laCv ? "" : tenLoai(vb.loai).toLocaleUpperCase("vi-VN"),
    trichYeu: laCv || !vb.trich_yeu ? [] : [vb.trich_yeu],
    than: nd.split(/\r?\n/).map((s) => s.trim()).filter(Boolean),
    noiNhan: vb.noi_nhan ?? [],
    kyTen: [vb.chuc_danh?.toLocaleUpperCase("vi-VN") ?? "", vb.nguoi_ky ?? ""].filter(Boolean),
  };
}

import { AlignmentType, BorderStyle, Document, Packer, Paragraph, Table, TableCell, TableRow, TextRun, WidthType } from "docx";
import { laHoa, type BoCuc } from "./bo-cuc";

const PT = (n: number) => n * 2; // nửa điểm
const MM = (n: number) => Math.round(n * 56.693); // twip
const RONG = 11906 - MM(30) - MM(15);
const COT_TRAI = MM(65);
const COT_PHAI = RONG - COT_TRAI;
const THUT = 720; // 1,27 cm
const KHONG = { style: BorderStyle.NONE, size: 0, color: "FFFFFF" };
const VIEN_KHONG = { top: KHONG, bottom: KHONG, left: KHONG, right: KHONG };

type Kieu = { co?: number; dam?: boolean; nghieng?: boolean; canh?: (typeof AlignmentType)[keyof typeof AlignmentType]; truoc?: number; sau?: number; thut?: number };
const chu = (text: string, k: Kieu = {}) => new TextRun({ text, bold: k.dam, italics: k.nghieng, size: PT(k.co ?? 14), font: "Times New Roman" });
const doan = (text: string, k: Kieu = {}) => new Paragraph({
  alignment: k.canh ?? AlignmentType.CENTER, spacing: { before: k.truoc ?? 0, after: k.sau ?? 0, line: 288 },
  indent: k.thut ? { firstLine: k.thut } : undefined, children: [chu(text, k)],
});
/** Gạch liền ngắn căn giữa trong cột rộng `cot`, dài `dai` twip. */
const gach = (cot: number, dai: number) => new Paragraph({
  indent: { left: Math.max(0, (cot - dai) / 2), right: Math.max(0, (cot - dai) / 2) },
  border: { top: { style: BorderStyle.SINGLE, size: 6, color: "000000", space: 1 } }, spacing: { after: 60 }, children: [],
});
const o = (rong: number, children: Paragraph[]) => new TableCell({ width: { size: rong, type: WidthType.DXA }, borders: VIEN_KHONG, children: children.length ? children : [new Paragraph({})] });
const bang = (hang: TableCell[][]) => new Table({
  width: { size: RONG, type: WidthType.DXA }, columnWidths: [COT_TRAI, COT_PHAI],
  borders: { ...VIEN_KHONG, insideHorizontal: KHONG, insideVertical: KHONG },
  rows: hang.map((c) => new TableRow({ children: c })),
});

function bangDau(bc: BoCuc): Table {
  const trai: Paragraph[] = [];
  if (bc.coQuanChuQuan) trai.push(doan(bc.coQuanChuQuan.toLocaleUpperCase("vi-VN"), { co: 13 }));
  for (const d of bc.coQuan) trai.push(doan(d.toLocaleUpperCase("vi-VN"), { co: 13, dam: true }));
  if (trai.length) trai.push(gach(COT_TRAI, 1300));
  const phai = [doan("CỘNG HÒA XÃ HỘI CHỦ NGHĨA VIỆT NAM", { co: 13, dam: true }), doan("Độc lập - Tự do - Hạnh phúc", { co: 14, dam: true }), gach(COT_PHAI, 3500)];
  const so = bc.so ? (/^Số:\s*\//.test(bc.so) ? bc.so.replace(/^Số:\s*/, "Số:        ") : bc.so) : "Số:        /";
  const trai2 = [doan(so, { co: 13 })];
  if (bc.vv) trai2.push(doan(bc.vv, { co: 12, truoc: 60 }));
  const phai2 = [doan(bc.diaDanhNgay || "......, ngày ... tháng ... năm ......", { co: 14, nghieng: true })];

import { useCallback, useEffect, useRef, useState } from "react";
import { tacVu, type KetQuaTacVu, type TenTacVu, type UocTinh } from "@/api/tac-vu";
import { LoiApi } from "@/api/khach";
import { LOI_HUY, loiTacVu, MA_CHAN_CHAY, type LoiTacVu } from "./thong-bao-loi";

export type UocTinhHien = UocTinh | null | { loi: LoiTacVu; chan: boolean };
export type TrangThaiTacVu<T> =
  | { loai: "nghi" } | { loai: "dang-chay"; batDau: number } | { loai: "xong"; kq: KetQuaTacVu<T> } | { loai: "loi"; loi: LoiTacVu };

const TRE_MS = 600;

export function useTacVu<T>(ten: TenTacVu) {
  const [uoc, setUoc] = useState<UocTinhHien>(null);
  const [tt, setTt] = useState<TrangThaiTacVu<T>>({ loai: "nghi" });
  const chayRef = useRef<AbortController | null>(null);
  const uocRef = useRef<AbortController | null>(null);
  const henRef = useRef<ReturnType<typeof setTimeout> | null>(null);

  const uocTinhTre = useCallback((than: unknown | null) => {
    if (henRef.current) clearTimeout(henRef.current);
    uocRef.current?.abort();
    if (than === null) { setUoc(null); return; }
    henRef.current = setTimeout(() => {
      const c = new AbortController(); uocRef.current = c;
      tacVu.uocTinh(ten, than, c.signal).then(
        (u) => { if (!c.signal.aborted) setUoc(u); },
        (e) => { if (!c.signal.aborted) setUoc({ loi: loiTacVu(e), chan: e instanceof LoiApi && MA_CHAN_CHAY.has(e.code) }); },
      );
    }, TRE_MS);
  }, [ten]);

  const chay = useCallback(async (than: unknown): Promise<KetQuaTacVu<T> | null> => {
    if (chayRef.current) return null; // đang chạy: bỏ qua bấm đôi
    const c = new AbortController(); chayRef.current = c;
    setTt({ loai: "dang-chay", batDau: Date.now() });
    try {
      const kq = await tacVu.chay<T>(ten, than, c.signal);
      if (chayRef.current !== c) return null; // lượt đã hủy / đã thay: bỏ kết quả muộn
      setTt({ loai: "xong", kq });
      setUoc((u) => (u && !("loi" in u) ? { ...u, so_du: kq.so_du, du_credit: kq.so_du >= u.credit_max } : u));
      return kq;
    } catch (e) {
      if (chayRef.current === c) setTt({ loai: "loi", loi: loiTacVu(e) });
      return null;
    } finally {
      if (chayRef.current === c) chayRef.current = null;
    }
  }, [ten]);

  /** Hủy: báo ngay, nhả khóa để chạy lại được kể cả khi yêu cầu cũ chưa kịp trả lỗi. */
  const huy = useCallback(() => {
    const c = chayRef.current;
    if (!c) return;
    chayRef.current = null;
    c.abort();
    setTt({ loai: "loi", loi: { thongBao: LOI_HUY, huy: true } });
  }, []);
  const datLai = useCallback(() => setTt({ loai: "nghi" }), []);
  useEffect(() => () => { chayRef.current?.abort(); uocRef.current?.abort(); if (henRef.current) clearTimeout(henRef.current); }, []);
  return { uoc, tt, uocTinhTre, chay, huy, datLai };
}

const bus = new EventTarget();

export const MIME_TRICH_XUAT = ["application/pdf", "image/png", "image/jpeg", "image/webp", "image/heic", "image/heif"];
const TOI_DA_TEP = 20;
const TOI_DA_BYTE = 20 * 1024 * 1024;
// Gói Free chỉ cho 5 MB mỗi lần trích xuất (subscription_plans.max_file_size_mb) → nén mọi ảnh > 800 KB.
// Trần theo gói do máy chủ báo (QUA_DAI kèm số MB) — web không ghi cứng.
const NGUONG_NEN = 800 * 1024;
const NEN_DUOC = new Set(["image/jpeg", "image/png", "image/webp"]);

export function kiemTepTrichXuat(ds: File[]): string | null {
  if (!ds.length) return "Chọn ít nhất một ảnh hoặc tệp PDF.";
  if (ds.length > TOI_DA_TEP) return "Tối đa 20 tệp mỗi lần.";
  for (const f of ds) if (!MIME_TRICH_XUAT.includes(doanMime(f))) return `Chưa hỗ trợ tệp ${f.name}. Dùng ảnh JPG, PNG, WEBP, HEIC hoặc PDF.`;
  const tong = ds.reduce((s, f) => s + f.size, 0);
  // ảnh lớn sẽ được thu nhỏ; chỉ chặn trước khi PDF / HEIC (không nén được) làm vượt trần
  const khongNen = ds.filter((f) => !NEN_DUOC.has(doanMime(f))).reduce((s, f) => s + f.size, 0);
  if (khongNen > TOI_DA_BYTE || (tong > TOI_DA_BYTE && khongNen === tong)) return "Tổng dung lượng tối đa 20 MB mỗi lần. Chia thành nhiều lần quét.";
  return null;
}

export function kichThuocMoi(w: number, h: number, toiDa = 2400) {
  const tl = Math.min(1, toiDa / Math.max(w, h));
  return { rong: Math.round(w * tl), cao: Math.round(h * tl) };
}

async function nenTrinhDuyet(f: File): Promise<Blob> {
  const anh = await createImageBitmap(f, { imageOrientation: "from-image" }); // ảnh điện thoại xoay theo EXIF
  const { rong, cao } = kichThuocMoi(anh.width, anh.height);
  const c = document.createElement("canvas"); c.width = rong; c.height = cao;
  const ctx = c.getContext("2d")!;
  ctx.fillStyle = "#fff"; ctx.fillRect(0, 0, rong, cao); // PNG trong suốt → JPEG nền trắng, không nền đen
  ctx.drawImage(anh, 0, 0, rong, cao);
  anh.close();
  return await new Promise<Blob>((ok, hong) => c.toBlob((b) => (b ? ok(b) : hong(new Error("Không thu nhỏ được ảnh."))), "image/jpeg", 0.85));
}

const docBase64 = (b: Blob) => new Promise<string>((ok, hong) => {
  const r = new FileReader();
  r.onload = () => ok(String(r.result).replace(/^data:[^,]*,/, ""));
  r.onerror = () => hong(r.error ?? new Error("Không đọc được tệp."));
  r.readAsDataURL(b);
});

export async function chuanBiTep(f: File, nen: (f: File) => Promise<Blob> = nenTrinhDuyet) {

  let b: Blob = f; let m = mime;
  if (NEN_DUOC.has(mime) && f.size > NGUONG_NEN) { b = await nen(f); m = "image/jpeg"; }
  return { base64: await docBase64(b), mime: m, byte: b.size };
}

export type CoQuan = { ten_co_quan: string; xa: string; tinh: string; dac_diem: string[]; nguoi_ky: string; chuc_danh: string };
const TRONG: CoQuan = { ten_co_quan: "", xa: "", tinh: "", dac_diem: [], nguoi_ky: "", chuc_danh: "" };
const khoa = (uid: string) => `vbp-co-quan:${uid}`;
const chuoi = (v: unknown, toiDa = 300) => (typeof v === "string" ? v.slice(0, toiDa) : "");

/** Lưu trên trình duyệt này theo tài khoản (đợt sau chuyển lên máy chủ — hợp đồng §9). */
export function docCoQuan(uid: string): CoQuan {
  try {
    const j = JSON.parse(localStorage.getItem(khoa(uid)) ?? "null") as Record<string, unknown> | null;
    if (!j || typeof j !== "object") return { ...TRONG };
    return { ten_co_quan: chuoi(j.ten_co_quan), xa: chuoi(j.xa), tinh: chuoi(j.tinh), nguoi_ky: chuoi(j.nguoi_ky), chuc_danh: chuoi(j.chuc_danh),
      dac_diem: Array.isArray(j.dac_diem) ? j.dac_diem.map((x) => chuoi(x)).slice(0, 10) : [] };
  } catch { return { ...TRONG }; }
}
export function luuCoQuan(uid: string, cq: CoQuan): boolean {
  try { localStorage.setItem(khoa(uid), JSON.stringify(cq)); return true; } catch { return false; }
}
/** dia_phuong của soan-thao: cần cả xã và tỉnh; dac_diem bỏ dòng trống, tối đa 10. */
export function diaPhuongTu(cq: CoQuan): { xa: string; tinh: string; dac_diem: string[] } | null {
  if (!cq.xa.trim() || !cq.tinh.trim()) return null;
  return { xa: cq.xa.trim(), tinh: cq.tinh.trim(), dac_diem: cq.dac_diem.map((s) => s.trim()).filter(Boolean).slice(0, 10) };
}

import { laHoa, phanTichVanBan } from "./xuat-word/bo-cuc";

/** «PHÓ CHỦ TỊCH» → «Phó Chủ tịch»: viết hoa chữ đầu, «chủ tịch» / «trưởng» … giữ như cách viết chức danh thông dụng. */
export function chucDanhDangCau(s: string): string {
  const t = s.toLocaleLowerCase("vi-VN").replace(/^(kt|tm|tl|tuq|q)\.\s*/u, "");
  return t.replace(/(^|\s)(phó|chủ|trưởng|giám|bí|chánh|hiệu|viện)(?=\s|$)/gu, (_m, d: string, w: string) => d + w[0].toLocaleUpperCase("vi-VN") + w.slice(1))
    .replace(/^\p{Ll}/u, (c) => c.toLocaleUpperCase("vi-VN"));
}
const laChoTrong = (s: string) => /^\[.*\]$/.test(s.trim());

export function thanVanBanDi(chu: string, loai: string, cq: CoQuan | null): Record<string, unknown> {
  const bc = phanTichVanBan(chu);
  const { so, ky_hieu } = tachSoKyHieu(bc.so);
  const ky = bc.kyTen.filter((s) => !laChoTrong(s));
  const ten = ky.length && !laHoa(ky[ky.length - 1]) ? ky[ky.length - 1] : "";
  const dongChuc = (ten ? ky.slice(0, -1) : ky).at(-1) ?? "";
  const chucDanh = dongChuc ? chucDanhDangCau(dongChuc) : "";
  const trichYeu = (bc.vv.replace(/^V\/v\s*/iu, "") || bc.trichYeu[0] || chu.trim().split(/\r?\n/)[0] || "Văn bản soạn bằng AI").slice(0, 2000);

    chuc_danh: chucDanh || cq?.chuc_danh.trim() || null, noi_dung: chu,
  };
}
