# L3 bản web VanBanPlus — API kho văn bản + dung lượng + phí lưu trữ: kế hoạch thực hiện

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Đưa lên production API kho văn bản cá nhân (thư mục, văn bản, tìm không dấu, tệp hai bước), dung lượng theo gói + gói dung lượng mua thêm (đơn `dung_luong` + nhánh `pay_order`), chế độ chỉ đọc khi vượt, dọn định kỳ và nhắc email.

**Architecture:** Mọi luật ở CSDL nằm trong hàm `SECURITY DEFINER` trả `jsonb` (bảng `kho_*` thu hồi mọi quyền, kể cả `service_role`), gọi qua `GoiRpc` (production: `supabase.rpc`, kiểm: `pg` thẳng) — cùng khuôn L1 (`src/lib/phien/**`). Tệp nằm ở bucket Storage riêng `kho`, đi qua giao diện `KhoTep` (production: storage-js, kiểm: bản giả trong bộ nhớ). Lớp dịch vụ `src/lib/kho/*.ts` trả `KetQua` của L1; route chỉ xác thực rồi gọi dịch vụ. Việc định kỳ: pg_cron → pg_net → `/api/noi-bo/*` (khóa bí mật `VBP_NOI_BO_KHOA`), vì xóa tệp Storage phải đi qua storage-api.

**Tech Stack:** Next.js 16 App Router, TypeScript, `@supabase/supabase-js` 2.95 (storage-js `createSignedUploadUrl`, `info`, `createSignedUrl`, `remove`), Postgres 17 (Supabase tự host) + `unaccent` + `pg_cron` + `pg_net` + Vault, bộ kiểm `node:test` qua `tsx`.

**Spec:** `docs/superpowers/specs/2026-11-17-ban-web-dot-a-design.md` (mục 5, 6, 8, 9) · hợp đồng giữa các luồng `docs/superpowers/plans/2026-11-17-hop-dong-web-dot-a.md` (mục 1, 2, 3, 5, 7).

import type { Client } from "pg";
import type { KhoTep } from "../../src/lib/kho/storage";

/** Tham số chủ sở hữu cá nhân cho các hàm kho_*. */
export const chuCua = (userId: string) => ({ p_chu_loai: "ca_nhan", p_chu_id: userId });

/** Đặt gói cho người dùng thử (het = null: không hạn). */
export async function datGoi(c: Client, userId: string, goi: string, het: Date | null): Promise<void> {
  await c.query(`UPDATE users SET subscription_plan_id = $2, subscription_end_date = $3 WHERE id = $1`, [
    userId, goi, het ? het.toISOString() : null,
  ]);
}

/** Ghi thẳng byte đã dùng (giả lập kho đã đầy). */
export async function datDaDung(c: Client, userId: string, byte: number): Promise<void> {
  await c.query(
    `INSERT INTO kho_dung_luong (chu_loai, chu_id, byte_da_dung) VALUES ('ca_nhan', $1, $2)
     ON CONFLICT (chu_loai, chu_id) DO UPDATE SET byte_da_dung = EXCLUDED.byte_da_dung`,
    [userId, byte]
  );
}

/** Storage giả trong bộ nhớ: tep = tệp «đã tải lên», daXoa = đường dẫn đã xóa. */
export function khoTepGia() {
  const tep = new Map<string, { byte: number; mime: string }>();

CREATE OR REPLACE FUNCTION kho_thu_muc_sua(p_chu_loai TEXT, p_chu_id UUID, p_id UUID, p_du_lieu JSONB) RETURNS JSONB
LANGUAGE plpgsql SECURITY DEFINER SET search_path = public, pg_temp AS $$

BEGIN
    PERFORM kho_kiem_chu(p_chu_loai, p_chu_id);
    SELECT * INTO m FROM kho_thu_muc WHERE id = p_id AND chu_loai = p_chu_loai AND chu_id = p_chu_id FOR UPDATE;
    IF NOT FOUND THEN RETURN jsonb_build_object('loi', 'KHONG_TIM_THAY'); END IF;
    v_cha_id := m.cha_id;
    IF p_du_lieu ? 'cha_id' THEN
        v_cha_id := (p_du_lieu->>'cha_id')::UUID;
        IF v_cha_id IS NOT NULL THEN
            SELECT * INTO v_cha FROM kho_thu_muc WHERE id = v_cha_id AND chu_loai = p_chu_loai AND chu_id = p_chu_id;
            IF NOT FOUND THEN RETURN jsonb_build_object('loi', 'KHONG_TIM_THAY'); END IF;
            IF v_cha.huong <> m.huong THEN RETURN kho_loi_dau_vao('Thư mục cha phải cùng hướng.'); END IF;
            IF EXISTS (SELECT 1 FROM kho_thu_muc_con(m.id) c WHERE c.con_id = v_cha_id) THEN

        END IF;

$$;

-- Sửa từng trường có trong p_du_lieu; p_phien_ban khác bản trong CSDL → XUNG_DOT_PHIEN_BAN kèm ban_moi.

-- Xóa hẳn một văn bản + mọi tệp gắn với nó; trừ byte tệp đã xác nhận. Trả đường dẫn Storage để API xóa tệp.
-- Người gọi đã kiểm chủ + thùng rác và đang giữ khóa dòng văn bản.
CREATE OR REPLACE FUNCTION kho_xoa_van_ban_noi(p_id UUID) RETURNS TEXT[]
LANGUAGE plpgsql SET search_path = public, pg_temp AS $$
DECLARE v_chu_loai TEXT; v_chu_id UUID; v_bot BIGINT; v_duong TEXT[];
BEGIN
    SELECT chu_loai, chu_id INTO v_chu_loai, v_chu_id FROM kho_van_ban WHERE id = p_id;
    PERFORM kho_khoa_dung_luong(v_chu_loai, v_chu_id);
    WITH x AS (DELETE FROM kho_tep WHERE gan_loai = 'van_ban' AND gan_id = p_id RETURNING duong_dan, trang_thai, byte)
    SELECT COALESCE(array_agg(duong_dan), '{}'), COALESCE(sum(byte) FILTER (WHERE trang_thai = 'xong'), 0)
      INTO v_duong, v_bot FROM x;
    UPDATE kho_dung_luong SET byte_da_dung = GREATEST(0, byte_da_dung - v_bot), sua_luc = now()
     WHERE chu_loai = v_chu_loai AND chu_id = v_chu_id;
    DELETE FROM kho_van_ban WHERE id = p_id;
    RETURN v_duong;
END;
$$;

CREATE OR REPLACE FUNCTION kho_van_ban_xoa_han(p_chu_loai TEXT, p_chu_id UUID, p_id UUID) RETURNS JSONB
LANGUAGE plpgsql SECURITY DEFINER SET search_path = public, pg_temp AS $$
DECLARE v_xoa TIMESTAMPTZ;
BEGIN
    PERFORM kho_kiem_chu(p_chu_loai, p_chu_id);
    SELECT xoa_luc INTO v_xoa FROM kho_van_ban WHERE id = p_id AND chu_loai = p_chu_loai AND chu_id = p_chu_id FOR UPDATE;
    IF NOT FOUND THEN RETURN jsonb_build_object('loi', 'KHONG_TIM_THAY'); END IF;
    IF v_xoa IS NULL THEN RETURN jsonb_build_object('loi', 'CHUA_O_THUNG_RAC'); END IF;
    RETURN jsonb_build_object('ok', TRUE, 'duong_dan', to_jsonb(kho_xoa_van_ban_noi(p_id)));
END;
$$;

-- p_loc: { tu_khoa: text[] (đã bỏ dấu, chỉ a-z0-9), huong?, loai?, tu?, den? (YYYY-MM-DD, theo ngay_ban_hanh),
--          thu_muc?, the?, sao?, thung_rac?, trang?, so_dong? (≤ 100) }. Mỗi từ khóa khớp tiền tố, mọi từ phải có.
CREATE OR REPLACE FUNCTION kho_van_ban_tim(p_chu_loai TEXT, p_chu_id UUID, p_loc JSONB) RETURNS JSONB
LANGUAGE plpgsql STABLE SECURITY DEFINER SET search_path = public, pg_temp AS $$
DECLARE
    v_q TSQUERY; v_so INT; v_trang INT; v_tong BIGINT; v_muc JSONB;
    v_rac BOOLEAN := COALESCE((p_loc->>'thung_rac')::BOOLEAN, FALSE);
BEGIN
    PERFORM kho_kiem_chu(p_chu_loai, p_chu_id);
    v_so := LEAST(GREATEST(COALESCE((p_loc->>'so_dong')::INT, 20), 1), 100);
    v_trang := LEAST(GREATEST(COALESCE((p_loc->>'trang')::INT, 1), 1), 10000);
    SELECT to_tsquery('simple', string_agg(quote_literal(t) || ':*', ' & ')) INTO v_q
      FROM jsonb_array_elements_text(CASE WHEN jsonb_typeof(p_loc->'tu_khoa') = 'array'
                                          THEN p_loc->'tu_khoa' ELSE '[]'::jsonb END) AS t
     WHERE t ~ '^[a-z0-9]{1,40}$';

    WITH loc AS (
        SELECT v.id, v.ngay_ban_hanh, v.tao_luc,
               CASE WHEN v_q IS NULL THEN 0 ELSE ts_rank(v.tim, v_q) END AS hang
          FROM kho_van_ban v
         WHERE v.chu_loai = p_chu_loai AND v.chu_id = p_chu_id
           AND (v.xoa_luc IS NOT NULL) = v_rac
           AND (v_q IS NULL OR v.tim @@ v_q)
           AND (p_loc->>'huong' IS NULL OR v.huong = p_loc->>'huong')
           AND (p_loc->>'loai' IS NULL OR v.loai = p_loc->>'loai')
           AND (p_loc->>'tu' IS NULL OR v.ngay_ban_hanh >= (p_loc->>'tu')::DATE)
           AND (p_loc->>'den' IS NULL OR v.ngay_ban_hanh <= (p_loc->>'den')::DATE)
           AND (p_loc->>'thu_muc' IS NULL OR v.thu_muc_id = (p_loc->>'thu_muc')::UUID)
           AND (p_loc->>'the' IS NULL OR v.the @> ARRAY[p_loc->>'the'])
           AND (p_loc->>'sao' IS NULL OR v.gan_sao = (p_loc->>'sao')::BOOLEAN)
    ), trang AS (
        SELECT id, hang, ngay_ban_hanh, tao_luc FROM loc
         ORDER BY hang DESC, ngay_ban_hanh DESC NULLS LAST, tao_luc DESC
         LIMIT v_so OFFSET (v_trang - 1) * v_so
    )
    SELECT (SELECT count(*) FROM loc),
           COALESCE((SELECT jsonb_agg(kho_van_ban_json(v, FALSE)
                                      ORDER BY tr.hang DESC, tr.ngay_ban_hanh DESC NULLS LAST, tr.tao_luc DESC)
                       FROM trang tr JOIN kho_van_ban v ON v.id = tr.id), '[]'::jsonb)
      INTO v_tong, v_muc;
    RETURN jsonb_build_object('muc', v_muc, 'tong', v_tong);
END;
$$;

DO $$
DECLARE f TEXT;
BEGIN
    FOREACH f IN ARRAY ARRAY[
        'kho_dung_luong_xem(text,uuid)', 'kho_thu_muc_ds(text,uuid)', 'kho_thu_muc_tao(text,uuid,uuid,jsonb)',
        'kho_thu_muc_sua(text,uuid,uuid,jsonb)', 'kho_thu_muc_xoa(text,uuid,uuid)',
        'kho_van_ban_tao(text,uuid,uuid,jsonb)', 'kho_van_ban_doc(text,uuid,uuid)', 'kho_van_ban_co(text,uuid,uuid)',
        'kho_van_ban_sua(text,uuid,uuid,integer,jsonb)', 'kho_van_ban_xoa_mem(text,uuid,uuid)',
        'kho_van_ban_khoi_phuc(text,uuid,uuid)', 'kho_van_ban_xoa_han(text,uuid,uuid)', 'kho_van_ban_tim(text,uuid,jsonb)'
    ] LOOP
        EXECUTE format('REVOKE ALL ON FUNCTION %s FROM PUBLIC, anon, authenticated', f);
        EXECUTE format('GRANT EXECUTE ON FUNCTION %s TO service_role', f);
    END LOOP;
    FOREACH f IN ARRAY ARRAY[
        'kho_kiem_chu(text,uuid)', 'kho_loi_dau_vao(text)', 'kho_mang(jsonb)', 'kho_tep_json(kho_tep)',
        'kho_van_ban_json(kho_van_ban,boolean)', 'kho_thu_muc_json(kho_thu_muc)', 'kho_han_muc(text,uuid)',
        'kho_khoa_dung_luong(text,uuid)', 'kho_thu_muc_cap(uuid)', 'kho_thu_muc_con(uuid)',
        'kho_van_ban_kiem_thu_muc(text,uuid,uuid,text)', 'kho_xoa_van_ban_noi(uuid)'
    ] LOOP
        EXECUTE format('REVOKE ALL ON FUNCTION %s FROM PUBLIC, anon, authenticated, service_role', f);
    END LOOP;
END;
$$;

-- ============================================================
-- Phần C: tệp (xin tải lên → Storage → xác nhận), xóa, dọn
-- ============================================================

-- Ghi hàng chờ 1 giờ. Hạn mức tính cả tệp đang chờ (chống tải song song vượt mức). Tối đa 20 tệp chờ mỗi chủ.
-- Kiểm chủ của đối tượng gắn (văn bản / cuộc họp / album) làm ở API (src/lib/kho/gan-tep.ts) trước khi gọi.
CREATE OR REPLACE FUNCTION kho_tep_xin(p_chu_loai TEXT, p_chu_id UUID, p_tao_boi UUID, p_gan_loai TEXT, p_gan_id UUID,
    p_ten TEXT, p_mime TEXT, p_byte BIGINT) RETURNS JSONB
LANGUAGE plpgsql SECURITY DEFINER SET search_path = public, pg_temp AS $$
DECLARE v_da BIGINT; v_cho BIGINT; v_so_cho INT; v_id UUID := gen_random_uuid(); v_duong TEXT;
BEGIN
    PERFORM kho_kiem_chu(p_chu_loai, p_chu_id);
    v_da := kho_khoa_dung_luong(p_chu_loai, p_chu_id);
    SELECT count(*), COALESCE(sum(byte_khai), 0) INTO v_so_cho, v_cho FROM kho_tep
     WHERE chu_loai = p_chu_loai AND chu_id = p_chu_id AND trang_thai = 'cho' AND het_han_cho > now();
    IF v_so_cho >= 20 THEN RETURN jsonb_build_object('loi', 'QUA_NHIEU_YEU_CAU'); END IF;
    IF v_da + v_cho + p_byte > (kho_han_muc(p_chu_loai, p_chu_id)->>'han_muc_byte')::BIGINT THEN
        RETURN jsonb_build_object('loi', 'HET_DUNG_LUONG');
    END IF;
    v_duong := p_chu_loai || '/' || p_chu_id::TEXT || '/' || v_id::TEXT;
    INSERT INTO kho_tep (id, chu_loai, chu_id, gan_loai, gan_id, duong_dan, ten, mime, byte_khai, trang_thai, het_han_cho, tao_boi)
    VALUES (v_id, p_chu_loai, p_chu_id, p_gan_loai, p_gan_id, v_duong, p_ten, p_mime, p_byte, 'cho',
            now() + interval '1 hour', p_tao_boi);
    RETURN jsonb_build_object('ok', TRUE, 'id', v_id, 'duong_dan', v_duong);
END;
$$;

CREATE OR REPLACE FUNCTION kho_tep_doc(p_chu_loai TEXT, p_chu_id UUID, p_id UUID) RETURNS JSONB
LANGUAGE sql STABLE SECURITY DEFINER SET search_path = public, pg_temp AS $$
    SELECT jsonb_build_object('id', t.id, 'duong_dan', t.duong_dan, 'ten', t.ten, 'mime', t.mime,
                              'byte_khai', t.byte_khai, 'trang_thai', t.trang_thai, 'tep', kho_tep_json(t))
      FROM kho_tep t WHERE t.id = p_id AND t.chu_loai = p_chu_loai AND t.chu_id = p_chu_id;
$$;

-- p_byte: kích thước thật API đọc từ Storage. Gọi lại khi đã xác nhận: trả lại tệp, không cộng đôi.
CREATE OR REPLACE FUNCTION kho_tep_xac_nhan(p_chu_loai TEXT, p_chu_id UUID, p_id UUID, p_byte BIGINT) RETURNS JSONB
LANGUAGE plpgsql SECURITY DEFINER SET search_path = public, pg_temp AS $$
DECLARE v_da BIGINT; t kho_tep%ROWTYPE;
BEGIN
    PERFORM kho_kiem_chu(p_chu_loai, p_chu_id);
    v_da := kho_khoa_dung_luong(p_chu_loai, p_chu_id);
    SELECT * INTO t FROM kho_tep WHERE id = p_id AND chu_loai = p_chu_loai AND chu_id = p_chu_id FOR UPDATE;
    IF NOT FOUND THEN RETURN jsonb_build_object('loi', 'KHONG_TIM_THAY'); END IF;
    IF t.trang_thai = 'xong' THEN RETURN jsonb_build_object('ok', TRUE, 'tep', kho_tep_json(t)); END IF;
    IF v_da + p_byte > (kho_han_muc(p_chu_loai, p_chu_id)->>'han_muc_byte')::BIGINT THEN
        RETURN jsonb_build_object('loi', 'HET_DUNG_LUONG');
    END IF;
    UPDATE kho_tep SET trang_thai = 'xong', byte = p_byte, het_han_cho = NULL, xac_nhan_luc = now(), sua_luc = now(),
                       phien_ban = phien_ban + 1
     WHERE id = t.id RETURNING * INTO t;
    UPDATE kho_dung_luong SET byte_da_dung = byte_da_dung + p_byte, sua_luc = now()
     WHERE chu_loai = p_chu_loai AND chu_id = p_chu_id;
    RETURN jsonb_build_object('ok', TRUE, 'tep', kho_tep_json(t));
END;
$$;

-- Xóa một tệp (đã xác nhận hoặc đang chờ); trừ byte nếu đã xác nhận. Trả đường dẫn để API xóa trên Storage.
CREATE OR REPLACE FUNCTION kho_tep_xoa(p_chu_loai TEXT, p_chu_id UUID, p_id UUID) RETURNS JSONB
LANGUAGE plpgsql SECURITY DEFINER SET search_path = public, pg_temp AS $$
DECLARE t kho_tep%ROWTYPE;
BEGIN
    PERFORM kho_kiem_chu(p_chu_loai, p_chu_id);
    PERFORM kho_khoa_dung_luong(p_chu_loai, p_chu_id);
    DELETE FROM kho_tep WHERE id = p_id AND chu_loai = p_chu_loai AND chu_id = p_chu_id RETURNING * INTO t;
    IF NOT FOUND THEN RETURN jsonb_build_object('loi', 'KHONG_TIM_THAY'); END IF;
    IF t.trang_thai = 'xong' THEN
        UPDATE kho_dung_luong SET byte_da_dung = GREATEST(0, byte_da_dung - t.byte), sua_luc = now()
         WHERE chu_loai = p_chu_loai AND chu_id = p_chu_id;
    END IF;
    RETURN jsonb_build_object('ok', TRUE, 'duong_dan', t.duong_dan);
END;
$$;

-- Hàng chờ quá hạn (tối đa 500 dòng mỗi lần). Tệp chờ chưa cộng byte nên không đổi dung lượng.
CREATE OR REPLACE FUNCTION kho_don_tep_cho() RETURNS JSONB
LANGUAGE sql SECURITY DEFINER SET search_path = public, pg_temp AS $$
    WITH x AS (
        DELETE FROM kho_tep WHERE id IN (
            SELECT id FROM kho_tep WHERE trang_thai = 'cho' AND het_han_cho < now()
             ORDER BY het_han_cho LIMIT 500 FOR UPDATE SKIP LOCKED)
        RETURNING duong_dan)
    SELECT jsonb_build_object('duong_dan', COALESCE(jsonb_agg(duong_dan), '[]'::jsonb)) FROM x;
$$;

-- Văn bản ở thùng rác quá p_ngay ngày: xóa hẳn (tối đa 200 văn bản mỗi lần).
CREATE OR REPLACE FUNCTION kho_don_thung_rac(p_ngay INT) RETURNS JSONB
LANGUAGE plpgsql SECURITY DEFINER SET search_path = public, pg_temp AS $$
DECLARE r RECORD; v_tat TEXT[] := '{}'; n INT := 0;
BEGIN
    IF p_ngay IS NULL OR p_ngay < 1 THEN RAISE EXCEPTION 'kho_don_thung_rac: số ngày sai'; END IF;
    FOR r IN SELECT id FROM kho_van_ban WHERE xoa_luc < now() - make_interval(days => p_ngay)
              ORDER BY xoa_luc LIMIT 200 FOR UPDATE SKIP LOCKED LOOP
        v_tat := v_tat || kho_xoa_van_ban_noi(r.id);
        n := n + 1;
    END LOOP;
    RETURN jsonb_build_object('van_ban', n, 'duong_dan', to_jsonb(v_tat));
END;
$$;

DO $$
DECLARE f TEXT;
BEGIN
    FOREACH f IN ARRAY ARRAY[
        'kho_tep_xin(text,uuid,uuid,text,uuid,text,text,bigint)', 'kho_tep_doc(text,uuid,uuid)',
        'kho_tep_xac_nhan(text,uuid,uuid,bigint)', 'kho_tep_xoa(text,uuid,uuid)', 'kho_don_tep_cho()',
        'kho_don_thung_rac(integer)'
    ] LOOP
        EXECUTE format('REVOKE ALL ON FUNCTION %s FROM PUBLIC, anon, authenticated', f);
        EXECUTE format('GRANT EXECUTE ON FUNCTION %s TO service_role', f);
    END LOOP;
END;
$$;

-- Cấp gói dung lượng cho đơn ĐÃ trả. Gọi lại trả lại dòng cũ. Đơn chưa trả / không phải đơn dung lượng: NULL.
-- SECURITY DEFINER: pay_order chạy bằng service_role, còn kho_dung_luong_mua đã thu hồi quyền của service_role.
CREATE OR REPLACE FUNCTION kho_cap_dung_luong(p_order UUID) RETURNS JSONB
LANGUAGE plpgsql SECURITY DEFINER SET search_path = public, pg_temp AS $$
DECLARE o orders%ROWTYPE; g JSONB; v_mb INT; v_goc TIMESTAMPTZ; v_bat_dau TIMESTAMPTZ; m kho_dung_luong_mua%ROWTYPE;
BEGIN
    SELECT * INTO o FROM orders WHERE id = p_order;
    IF NOT FOUND OR o.loai <> 'dung_luong' OR o.status <> 'paid' THEN RETURN NULL; END IF;
    PERFORM 1 FROM users WHERE id = o.user_id FOR NO KEY UPDATE;

    SELECT * INTO m FROM kho_dung_luong_mua WHERE order_id = o.id;
    IF NOT FOUND THEN
        SELECT gia_tri INTO g FROM credit_bang_gia WHERE loai = 'goi_dung_luong' AND ma = o.goi_dung_luong_ma;
        IF g IS NULL THEN
            RAISE EXCEPTION 'kho_cap_dung_luong: không có gói dung lượng % trong bảng giá', o.goi_dung_luong_ma;
        END IF;
        v_mb := (g->>'them_mb')::INT;
        IF v_mb IS NULL OR v_mb <= 0 THEN
            RAISE EXCEPTION 'kho_cap_dung_luong: gói % sai (them_mb %)', o.goi_dung_luong_ma, v_mb;
        END IF;
        v_goc := COALESCE(o.paid_at, now());
        SELECT GREATEST(v_goc, COALESCE(max(het_han), v_goc)) INTO v_bat_dau FROM kho_dung_luong_mua
         WHERE chu_loai = 'ca_nhan' AND chu_id = o.user_id AND goi_ma = o.goi_dung_luong_ma AND het_han > v_goc;
        INSERT INTO kho_dung_luong_mua (chu_loai, chu_id, goi_ma, them_mb, bat_dau, het_han, order_id)
        VALUES ('ca_nhan', o.user_id, o.goi_dung_luong_ma, v_mb, v_bat_dau,
                v_bat_dau + make_interval(months => o.so_thang), o.id)
        ON CONFLICT (order_id) DO NOTHING;
        SELECT * INTO m FROM kho_dung_luong_mua WHERE order_id = o.id;
    END IF;
    RETURN jsonb_build_object('mua_id', m.id, 'them_mb', m.them_mb, 'bat_dau', m.bat_dau, 'het_han', m.het_han);
END;
$$;

-- pay_order: giữ nguyên bản 005, thêm nhánh 'dung_luong'.
CREATE OR REPLACE FUNCTION pay_order(p_code TEXT, p_source TEXT, p_ref TEXT)
RETURNS TEXT
LANGUAGE plpgsql
SET search_path = public
AS $$
DECLARE
    o orders%ROWTYPE;
    u users%ROWTYPE;
    v_extend BOOLEAN;
    v_start TIMESTAMPTZ;
    v_end TIMESTAMPTZ;
    v_lo JSONB;
    v_mua JSONB;
BEGIN
    SELECT * INTO o FROM orders WHERE code = p_code FOR UPDATE;
    IF NOT FOUND THEN RETURN 'not_found'; END IF;
    IF o.status = 'paid' THEN RETURN 'already_paid'; END IF;
    IF o.status = 'cancelled' AND p_source <> 'admin' THEN RETURN 'cancelled'; END IF;

    SELECT * INTO u FROM users WHERE id = o.user_id FOR NO KEY UPDATE;

    IF o.loai = 'nap' THEN
        UPDATE orders
           SET status = 'paid', paid_source = p_source, paid_ref = p_ref, paid_at = NOW(), period_start = NOW()
         WHERE id = o.id;
        v_lo := credit_cap_nap(o.id);
        UPDATE orders SET period_end = (v_lo->>'het_han')::TIMESTAMPTZ WHERE id = o.id;
        RETURN 'paid';
    END IF;

    IF o.loai = 'dung_luong' THEN
        UPDATE orders SET status = 'paid', paid_source = p_source, paid_ref = p_ref, paid_at = NOW() WHERE id = o.id;
        v_mua := kho_cap_dung_luong(o.id);
        UPDATE orders SET period_start = (v_mua->>'bat_dau')::TIMESTAMPTZ, period_end = (v_mua->>'het_han')::TIMESTAMPTZ
         WHERE id = o.id;
        RETURN 'paid';
    END IF;

    v_extend := u.subscription_plan_id = o.plan_id AND COALESCE(u.subscription_end_date > NOW(), FALSE);
    v_start := CASE WHEN v_extend THEN u.subscription_end_date ELSE NOW() END;
    v_end := v_start + CASE o.cycle WHEN 'year' THEN INTERVAL '1 year' ELSE INTERVAL '1 month' END;

    UPDATE users
       SET subscription_plan_id = o.plan_id,
           subscription_start_date = CASE WHEN v_extend THEN u.subscription_start_date ELSE v_start END,
           subscription_end_date = v_end
     WHERE id = u.id;

    UPDATE orders
       SET status = 'paid', paid_source = p_source, paid_ref = p_ref, paid_at = NOW(),
           period_start = v_start, period_end = v_end
     WHERE id = o.id;

    RETURN 'paid';
END;
$$;

-- Hạn mức = gói hiệu lực + gói dung lượng đang trong hạn (đã bắt đầu). goi_them liệt kê cả gói nối tiếp chưa bắt đầu.
CREATE OR REPLACE FUNCTION kho_han_muc(p_chu_loai TEXT, p_chu_id UUID) RETURNS JSONB
LANGUAGE plpgsql STABLE SET search_path = public, pg_temp AS $$
DECLARE v_goi TEXT; v_het TIMESTAMPTZ; v_mb INT := 0; v_them INT; v_ds JSONB;
BEGIN
    IF p_chu_loai = 'ca_nhan' THEN
        SELECT subscription_plan_id, subscription_end_date INTO v_goi, v_het FROM users WHERE id = p_chu_id;
        IF FOUND THEN
            IF v_het IS NOT NULL AND v_het < now() THEN v_goi := 'free'; END IF;
            SELECT cloud_storage_mb INTO v_mb FROM subscription_plans WHERE id = v_goi;
            v_mb := COALESCE(v_mb, 0);
        END IF;
    END IF;
    SELECT COALESCE(sum(them_mb) FILTER (WHERE bat_dau <= now()), 0),
           COALESCE(jsonb_agg(jsonb_build_object('ma', goi_ma, 'them_mb', them_mb, 'bat_dau', bat_dau, 'het_han', het_han)
                              ORDER BY het_han), '[]'::jsonb)
      INTO v_them, v_ds
      FROM kho_dung_luong_mua WHERE chu_loai = p_chu_loai AND chu_id = p_chu_id AND het_han > now();
    RETURN jsonb_build_object('goi_mb', v_mb, 'them_mb', v_them, 'han_muc_byte', (v_mb + v_them)::BIGINT * 1048576,
                              'goi_them', v_ds);
END;
$$;

-- Cập nhật chi_doc_tu rồi trả danh sách thư cần gửi (chưa có trong nhac_email), tối đa 500, kèm tổng byte toàn hệ thống.
-- khoa gắn với sự kiện: gói theo ngày hết hạn, gói dung lượng theo id, chỉ đọc theo mốc chi_doc_tu.
CREATE OR REPLACE FUNCTION kho_nhac_can_gui() RETURNS JSONB
LANGUAGE plpgsql SECURITY DEFINER SET search_path = public, pg_temp AS $$
DECLARE v_thu JSONB; v_tong BIGINT;
BEGIN
    UPDATE kho_dung_luong d SET chi_doc_tu = x.moi
      FROM (SELECT chu_loai, chu_id,
                   CASE WHEN byte_da_dung > (kho_han_muc(chu_loai, chu_id)->>'han_muc_byte')::BIGINT
                        THEN COALESCE(chi_doc_tu, now()) END AS moi
              FROM kho_dung_luong WHERE chu_loai = 'ca_nhan') x
     WHERE d.chu_loai = x.chu_loai AND d.chu_id = x.chu_id AND d.chi_doc_tu IS DISTINCT FROM x.moi;

    SELECT COALESCE(jsonb_agg(to_jsonb(x)), '[]'::jsonb) INTO v_thu FROM (
        SELECT * FROM (
            SELECT 'goi_sap_het'::TEXT AS loai,
                   u.id::TEXT || ':' || to_char(u.subscription_end_date AT TIME ZONE 'Asia/Ho_Chi_Minh', 'YYYY-MM-DD') AS khoa,
                   u.id AS user_id, u.email, u.full_name AS ho_ten,
                   jsonb_build_object('goi', p.name, 'het_han', u.subscription_end_date) AS chi_tiet
              FROM users u JOIN subscription_plans p ON p.id = u.subscription_plan_id
             WHERE u.is_active AND u.subscription_plan_id <> 'free'
               AND u.subscription_end_date > now() AND u.subscription_end_date <= now() + interval '7 days'
            UNION ALL
            SELECT 'dung_luong_sap_het', m.id::TEXT, u.id, u.email, u.full_name,
                   jsonb_build_object('goi_ma', m.goi_ma, 'them_mb', m.them_mb, 'het_han', m.het_han)
              FROM kho_dung_luong_mua m JOIN users u ON u.id = m.chu_id
             WHERE m.chu_loai = 'ca_nhan' AND u.is_active
               AND m.het_han > now() AND m.het_han <= now() + interval '7 days'
               AND NOT EXISTS (SELECT 1 FROM kho_dung_luong_mua m2
                                WHERE m2.chu_loai = m.chu_loai AND m2.chu_id = m.chu_id AND m2.goi_ma = m.goi_ma
                                  AND m2.het_han > m.het_han)
            UNION ALL
            SELECT CASE WHEN d.chi_doc_tu <= now() - interval '30 days' THEN 'chi_doc_30_ngay' ELSE 'vao_chi_doc' END,
                   d.chu_id::TEXT || ':' || floor(extract(epoch FROM d.chi_doc_tu))::BIGINT::TEXT, u.id, u.email,
                   u.full_name,
                   jsonb_build_object('da_dung_byte', d.byte_da_dung,
                                      'han_muc_byte', (kho_han_muc(d.chu_loai, d.chu_id)->>'han_muc_byte')::BIGINT,
                                      'tu', d.chi_doc_tu)
              FROM kho_dung_luong d JOIN users u ON u.id = d.chu_id
             WHERE d.chu_loai = 'ca_nhan' AND u.is_active AND d.chi_doc_tu IS NOT NULL
        ) t
        WHERE NOT EXISTS (SELECT 1 FROM nhac_email n WHERE n.loai = t.loai AND n.khoa = t.khoa)
        LIMIT 500
    ) x;

    SELECT COALESCE(sum(byte_da_dung), 0) INTO v_tong FROM kho_dung_luong;
    RETURN jsonb_build_object('thu', v_thu, 'tong_byte', v_tong);
END;
$$;

-- Giữ chỗ trước khi gửi (chống gửi đôi khi chạy chồng). moi = FALSE: đã có người giữ / đã gửi.
CREATE OR REPLACE FUNCTION kho_nhac_ghi(p_loai TEXT, p_khoa TEXT, p_user UUID) RETURNS JSONB
LANGUAGE plpgsql SECURITY DEFINER SET search_path = public, pg_temp AS $$

BEGIN

    ON CONFLICT (loai, khoa) DO NOTHING RETURNING id INTO v;
    RETURN jsonb_build_object('moi', v IS NOT NULL);
END;
$$;

-- Gửi hỏng: bỏ chỗ đã giữ để lần chạy sau gửi lại.
CREATE OR REPLACE FUNCTION kho_nhac_bo(p_loai TEXT, p_khoa TEXT) RETURNS JSONB
LANGUAGE sql SECURITY DEFINER SET search_path = public, pg_temp AS $$
    DELETE FROM nhac_email WHERE loai = p_loai AND khoa = p_khoa;
    SELECT jsonb_build_object('ok', TRUE);
$$;

DO $$
DECLARE f TEXT;
BEGIN
    FOREACH f IN ARRAY ARRAY['kho_cap_dung_luong(uuid)', 'pay_order(text,text,text)', 'kho_nhac_can_gui()',
                             'kho_nhac_ghi(text,text,uuid)', 'kho_nhac_bo(text,text)'] LOOP
        EXECUTE format('REVOKE ALL ON FUNCTION %s FROM PUBLIC, anon, authenticated', f);
        EXECUTE format('GRANT EXECUTE ON FUNCTION %s TO service_role', f);

END;
$$;

export const MIME_CHO_PHEP = [
  "application/pdf",
  "application/msword",
  "application/vnd.openxmlformats-officedocument.wordprocessingml.document",

  "application/vnd.openxmlformats-officedocument.spreadsheetml.sheet",

export type KQ<T> = { du_lieu: T } | { loi: string; code: "DAU_VAO_SAI" | "TEP_KHONG_HOP_LE" };

class LoiDauVao extends Error {
  constructor(message: string, readonly code: "DAU_VAO_SAI" | "TEP_KHONG_HOP_LE" = "DAU_VAO_SAI") {
    super(message);
  }
}
const sai = (m: string, code?: "DAU_VAO_SAI" | "TEP_KHONG_HOP_LE"): never => {
  throw new LoiDauVao(m, code);
};
function bat<T>(fn: () => T): KQ<T> {
  try {
    return { du_lieu: fn() };
  } catch (e) {
    if (e instanceof LoiDauVao) return { loi: e.message, code: e.code };
    throw e;
  }
}

const UUID = /^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i;
export const laUuid = (v: unknown): v is string => typeof v === "string" && UUID.test(v);

/** Chữ thường, bỏ dấu (cả đ). Khớp kho_khong_dau ở 010_kho.sql. */
export function boDau(s: string): string {
  return s.normalize("NFD").replace(/\p{M}+/gu, "").replace(/đ/g, "d").replace(/Đ/g, "D").toLowerCase();
}

export function tachTuKhoa(q: string): string[] {
  const ds = boDau(q).split(/[^a-z0-9]+/).filter(Boolean).map((t) => t.slice(0, 40));
  return [...new Set(ds)].slice(0, 8);
}

export function lamSachTenTep(ten: string): string {
  let s = (ten.split(/[\\/]/).pop() ?? "")
    .replace(/[\u0000-\u001f\u007f]/g, "")
    .replace(/"/g, "'")
    .replace(/\s+/g, " ")
    .trim()
    .replace(/^\.+/, "");
  if (s.length > 200) {
    const duoi = /\.[A-Za-z0-9]{1,10}$/.exec(s)?.[0] ?? "";
    s = s.slice(0, 200 - duoi.length) + duoi;
  }
  return s || "tep";
}

export const laXemTaiCho = (mime: string) => MIME_XEM_TAI_CHO.has(mime);

const coNul = (s: string) => s.includes("\u0000");

function chuoiTuy(v: unknown, nhan: string, toiDa: number): string | null {
  if (v === null || v === undefined) return null;
  if (typeof v !== "string") sai(`${nhan} phải là chữ.`);
  const s = (v as string).replace(/\u0000/g, "").trim();
  if (s.length > toiDa) sai(`${nhan} dài quá ${toiDa.toLocaleString("vi-VN")} ký tự.`);
  return s || null;
}

function ngay(v: unknown, nhan: string): string | null {
  if (v === null || v === undefined || v === "") return null;
  if (typeof v !== "string" || !/^\d{4}-\d{2}-\d{2}$/.test(v)) sai(`${nhan} phải có dạng YYYY-MM-DD.`);
  const d = new Date(`${v}T00:00:00Z`);
  if (Number.isNaN(d.getTime()) || d.toISOString().slice(0, 10) !== v) sai(`${nhan} không phải ngày có thật.`);
  return v as string;
}

function soNguyen(v: unknown, nhan: string, min: number, max: number): number | null {
  if (v === null || v === undefined) return null;
  if (typeof v !== "number" || !Number.isInteger(v) || v < min || v > max) sai(`${nhan} phải là số nguyên từ ${min} đến ${max}.`);
  return v as number;
}

function mangChuoi(v: unknown, nhan: string, toiDaPhanTu: number, toiDaKyTu: number): string[] {
  if (v === null || v === undefined) return [];
  if (!Array.isArray(v)) sai(`${nhan} phải là danh sách.`);
  const ds: string[] = [];
  for (const x of v as unknown[]) {
    const s = chuoiTuy(x, nhan, toiDaKyTu);
    if (s && !ds.includes(s)) ds.push(s);
  }
  if (ds.length > toiDaPhanTu) sai(`${nhan} tối đa ${toiDaPhanTu} mục.`);
  return ds;
}

function motTrong<T extends string>(v: unknown, nhan: string, ds: readonly T[]): T {
  if (typeof v !== "string" || !(ds as readonly string[]).includes(v)) sai(`${nhan} không hợp lệ.`);
  return v as T;
}

export function kiemVanBan(than: Record<string, unknown>, cheDo: "tao" | "sua"): KQ<Record<string, unknown>> {
  return bat(() => {
    const d: Record<string, unknown> = {};
    const co = (k: string) => Object.hasOwn(than, k);
    if (cheDo === "tao" || co("huong")) d.huong = motTrong(than.huong, "Hướng văn bản", HUONG);
    if (cheDo === "tao" || co("trich_yeu")) {
      const t = chuoiTuy(than.trich_yeu, "Trích yếu", 2000);
      if (!t) sai("Thiếu trích yếu.");
      d.trich_yeu = t;
    }
    for (const [k, nhan] of Object.entries(TRUONG_NGAN)) if (co(k)) d[k] = chuoiTuy(than[k], nhan, 500);
    for (const [k, [nhan, toiDa]] of Object.entries(TRUONG_DAI)) if (co(k)) d[k] = chuoiTuy(than[k], nhan, toiDa);
    for (const [k, nhan] of Object.entries(TRUONG_NGAY)) if (co(k)) d[k] = ngay(than[k], nhan);
    for (const [k, [nhan, n, toiDa]] of Object.entries(TRUONG_MANG)) if (co(k)) d[k] = mangChuoi(than[k], nhan, n, toiDa);
    if (co("the")) d.the = mangChuoi(than.the, "Thẻ", 30, 50);
    if (co("so_den")) d.so_den = soNguyen(than.so_den, "Số đến", 0, 1_000_000_000);
    if (co("do_khan")) d.do_khan = than.do_khan == null ? "thuong" : motTrong(than.do_khan, "Độ khẩn", DO_KHAN);
    if (co("do_mat")) d.do_mat = than.do_mat == null ? "thuong" : motTrong(than.do_mat, "Độ mật", DO_MAT);
    if (co("gan_sao")) {
      if (typeof than.gan_sao !== "boolean") sai("Gắn sao phải là true hoặc false.");
      d.gan_sao = than.gan_sao;
    }
    if (co("thu_muc_id")) {
      if (than.thu_muc_id !== null && !laUuid(than.thu_muc_id)) sai("Thư mục không hợp lệ.");
      d.thu_muc_id = than.thu_muc_id ?? null;
    }

export interface GoiDungLuong {

function soNguyenDuong(v: unknown, ten: string): number {

export function kiemGoiDungLuong(ma: string, gia_tri: unknown): GoiDungLuong {

}

/** Số tháng mua: số nguyên 1–12 (nhận cả chuỗi «3» từ tham số trang). Sai → null. */
export function kiemSoThang(v: unknown): number | null {
  const n = typeof v === "number" ? v : typeof v === "string" && /^\d{1,2}$/.test(v) ? Number(v) : NaN;
  return Number.isInteger(n) && n >= 1 && n <= 12 ? n : null;
}

  if (error) throw new Error(`Không đọc được gói dung lượng: ${error.message}`);
  const ds: GoiDungLuong[] = [];
  for (const d of (data ?? []) as Array<{ ma: string; gia_tri: unknown }>) {

    }
  }
  return ds.sort((a, b) => a.gia_vnd_thang - b.gia_vnd_thang);
}

  return l?.status === 404 || String(l?.statusCode ?? "") === "404" || /not.?found/i.test(String(l?.message ?? ""));
};

/** Bản production: bucket «kho» qua storage-js (khóa service). Đọc getSupabase lúc gọi, không lúc nạp mô-đun. */
export function khoTepSupabase(): KhoTep {
  const b = () => getSupabase().storage.from(BUCKET_KHO);
  return {
    async kyTaiLen(duongDan) {
      const { data, error } = await b().createSignedUploadUrl(duongDan);
      if (error || !data) throw new Error(`Storage ký tải lên: ${error?.message ?? "rỗng"}`);
      return { url: data.signedUrl };
    },
    async thongTin(duongDan) {
      const { data, error } = await b().info(duongDan);
      if (error) {
        if (laKhongCo(error)) return null;
        throw new Error(`Storage đọc thông tin: ${error.message}`);
      }
      const d = data as unknown as { size?: unknown; contentType?: unknown; metadata?: { size?: unknown; mimetype?: unknown } };
      const byte = Number(d.size ?? d.metadata?.size);
      const mime = String(d.contentType ?? d.metadata?.mimetype ?? "");
      if (!Number.isFinite(byte)) throw new Error("Storage đọc thông tin: thiếu kích thước");
      return { byte, mime };
    },
    async kyTaiVe(duongDan, giay, tenTai) {
      const { data, error } = await b().createSignedUrl(duongDan, giay, tenTai ? { download: tenTai } : undefined);
      if (error || !data) throw new Error(`Storage ký tải về: ${error?.message ?? "rỗng"}`);
      return data.signedUrl;
    },
    async xoa(duongDan) {
      for (let i = 0; i < duongDan.length; i += 100) {
        const { error } = await b().remove(duongDan.slice(i, i + 100));
        if (error) throw new Error(`Storage xóa: ${error.message}`);
      }
    },
  };
}

import { loi, type KetQua } from "../phien/dich-vu";
import type { KQ } from "./kiem-dau-vao";

export const xong = (data: unknown, message = "Thành công"): KetQua => ({ loai: "xong", message, data });

export const LOI_KHO = {
  khongTimThay: () => loi(404, "KHONG_TIM_THAY", "Không tìm thấy."),
  hetDungLuong: () =>
    loi(403, "HET_DUNG_LUONG", "Đã dùng hết dung lượng lưu trữ. Kho đang chỉ đọc: mua thêm dung lượng hoặc xóa bớt tệp."),
  tepKhongHopLe: (m: string) => loi(400, "TEP_KHONG_HOP_LE", m),
  tepChuaTaiLen: () => loi(409, "TEP_CHUA_TAI_LEN", "Chưa nhận được tệp. Tải lên xong rồi xác nhận lại."),
  dauVaoSai: (m: string) => loi(400, "DAU_VAO_SAI", m),
};

/** Kết quả lỗi của hàm SQL kho_* → KetQua. Mã lạ: ném lỗi (thành 500). */
export function tuLoiSql(r: { loi: string; thong_bao?: string; ban_moi?: unknown }): KetQua {
  switch (r.loi) {
    case "KHONG_TIM_THAY":
      return LOI_KHO.khongTimThay();
    case "HET_DUNG_LUONG":
      return LOI_KHO.hetDungLuong();
    case "XUNG_DOT_PHIEN_BAN":
      return loi(409, "XUNG_DOT_PHIEN_BAN", "Văn bản vừa được sửa ở nơi khác. Tải lại bản mới rồi sửa tiếp.", { ban_moi: r.ban_moi });
    case "THU_MUC_KHONG_TRONG":
      return loi(409, "THU_MUC_KHONG_TRONG", "Thư mục còn thư mục con hoặc văn bản.");
    case "CHUA_O_THUNG_RAC":
      return loi(409, "CHUA_O_THUNG_RAC", "Chỉ xóa hẳn được văn bản đã ở thùng rác.");
    case "DANG_O_THUNG_RAC":
      return loi(409, "DANG_O_THUNG_RAC", "Văn bản đang ở thùng rác. Khôi phục trước khi sửa.");
    case "QUA_NHIEU_YEU_CAU":
      return loi(429, "QUA_NHIEU_YEU_CAU", "Có quá nhiều tệp đang chờ tải lên. Thử lại sau ít phút.", { thu_lai_sau_giay: 60 });
    case "DAU_VAO_SAI":
      return LOI_KHO.dauVaoSai(r.thong_bao ?? "Dữ liệu không hợp lệ.");
    default:
      throw new Error(`kho: mã lỗi SQL lạ ${r.loi}`);
  }
}

/** Lỗi của hàm kiểm thuần → KetQua. */
export const tuKiem = (k: Extract<KQ<unknown>, { loi: string }>): KetQua =>
  k.code === "TEP_KHONG_HOP_LE" ? LOI_KHO.tepKhongHopLe(k.loi) : LOI_KHO.dauVaoSai(k.loi);

/** Đổi kết quả SQL { ok, <khoa> } | { loi } thành KetQua. */
export function tuKetQuaSql(r: any, khoa: string | null, message = "Thành công"): KetQua {
  if (r && typeof r === "object" && "loi" in r) return tuLoiSql(r);
  return xong(khoa ? r[khoa] : null, message);
}

/** Storage prod: UPLOAD_SIGNED_URL_EXPIRATION_TIME=120 — đường ký tải lên phải bắt đầu dùng trong 120 giây. */
export const GIAY_TAI_LEN = 120;
export const GIAY_TAI_VE = 600;

const sau = (pt: PhuThuocKho, giay: number) => new Date(pt.bayGio().getTime() + giay * 1000).toISOString();

/** Bỏ một tệp (hàng CSDL trước, rồi Storage; Storage lỗi chỉ ghi log). */

  }
}

export async function xinTaiLen(pt: PhuThuocKho, user: { id: string }, chu: Chu, than: Record<string, unknown>): Promise<KetQua> {
  const k = kiemXinTaiLen(than);
  if ("loi" in k) return tuKiem(k);
  const { ten, mime, byte, gan } = k.du_lieu;
  const kiem = layKiemGan(gan.loai);
  if (!kiem) return LOI_KHO.dauVaoSai("Chưa hỗ trợ gắn tệp vào loại này.");
  if (!(await kiem(pt.rpc, chu, gan.id))) return LOI_KHO.khongTimThay();
  const r = await pt.rpc("kho_tep_xin", {
    ...thamSoChu(chu), p_tao_boi: user.id, p_gan_loai: gan.loai, p_gan_id: gan.id, p_ten: ten, p_mime: mime, p_byte: byte,
  });
  if (r?.loi) return tuLoiSql(r);
  let url: string;
  try {
    url = (await pt.tep.kyTaiLen(r.duong_dan)).url;
  } catch (e) {
    await pt.rpc("kho_tep_xoa", { ...thamSoChu(chu), p_id: r.id });
    throw e;
  }
  return xong({ id: r.id, url_tai_len: url, het_han: sau(pt, GIAY_TAI_LEN) });
}

export async function xacNhanTep(pt: PhuThuocKho, chu: Chu, than: Record<string, unknown>): Promise<KetQua> {
  const id = than.id;
  if (!laUuid(id)) return LOI_KHO.khongTimThay();
  const t = await pt.rpc("kho_tep_doc", { ...thamSoChu(chu), p_id: id });
  if (!t) return LOI_KHO.khongTimThay();
  if (t.trang_thai === "xong") return xong(t.tep);
  const that = await pt.tep.thongTin(t.duong_dan);
  if (!that) return LOI_KHO.tepChuaTaiLen();
  const mimeThat = that.mime.split(";")[0].trim().toLowerCase();
  if (that.byte !== Number(t.byte_khai) || mimeThat !== t.mime) {
    await boTep(pt, chu, id, t.duong_dan);
    return LOI_KHO.tepKhongHopLe("Tệp tải lên khác thông tin đã khai báo (kích thước hoặc loại tệp). Tải lại tệp.");
  }
  const r = await pt.rpc("kho_tep_xac_nhan", { ...thamSoChu(chu), p_id: id, p_byte: that.byte });
  if (r?.loi) {
    if (r.loi === "HET_DUNG_LUONG") await boTep(pt, chu, id, t.duong_dan);
    return tuLoiSql(r);
  }
  return xong(r.tep, "Đã lưu tệp.");
}

  if (!t || t.trang_thai !== "xong") return LOI_KHO.khongTimThay();
  const url = await pt.tep.kyTaiVe(t.duong_dan, GIAY_TAI_VE, laXemTaiCho(t.mime) ? null : t.ten);
  return xong({ url, het_han: sau(pt, GIAY_TAI_VE) });
}

  } catch (e) {
    pt.ghiLog(`[KHO] xóa tệp: xóa Storage lỗi — ${(e as Error).message}`);
  }
  return xong(null, "Đã xóa tệp.");
}

import { NextResponse, type NextRequest } from "next/server";
import type { ApiUser } from "@/types";
import { LOI } from "../phien/dich-vu";
import { traLoi } from "../phien/tra-loi";
import { xacThucChiTiet } from "../tac-vu/xac-thuc";
import { chuCuaNguoiDung, type Chu } from "./chu-so-huu";

export { docThan, ngonCanh, traLoi } from "../phien/tra-loi";

/** Route kho: Bearer (web / app) hoặc X-API-Key. Lỗi xác thực trả đúng mã để web biết gọi lam-moi. */
export async function canNguoiDung(req: NextRequest): Promise<{ user: ApiUser; chu: Chu } | NextResponse> {
  const r = await xacThucChiTiet(req);
  if ("loi" in r) {
    return traLoi(r.loi === "PHIEN_HET_HAN" ? LOI.phienHetHan() : r.loi === "PHIEN_THU_HOI" ? LOI.phienThuHoi() : LOI.chuaDangNhap());
  }
  return { user: r.user, chu: chuCuaNguoiDung(r.user) };
}

export async function chay(fn: () => Promise<NextResponse>): Promise<NextResponse> {
  try {
    return await fn();
  } catch (e) {
    console.error("[KHO] lỗi bất ngờ", (e as Error).message);
    return NextResponse.json(
      { success: false, message: "Lỗi máy chủ. Thử lại sau.", code: "LOI_MAY_CHU" },
      { status: 500, headers: { "Cache-Control": "no-store" } }
    );
  }
}

export function tenDon(
  o: Pick<OrderRow, "loai" | "plan_id" | "goi_nap_ma"> & Partial<Pick<OrderRow, "goi_dung_luong_ma" | "so_thang">>,
  tenGoi: Record<string, string>,
  nap: GoiNap | null,
  dl: GoiDungLuong | null = null
): string {
  if (o.loai === "dung_luong") {
    const goi = dl ? `Thêm ${tenDungLuong(dl.them_mb)} dung lượng` : `Gói dung lượng ${o.goi_dung_luong_ma ?? ""}`.trim();
    return `${goi} · ${o.so_thang ?? 1} tháng`;

  if (o.loai === "nap") {
    return nap ? `Nạp ${nap.credit.toLocaleString("vi-VN")} credit` : `Gói nạp ${o.goi_nap_ma ?? ""}`.trim();
  }
  return (o.plan_id && tenGoi[o.plan_id]) || o.plan_id || "";
}

import { createHash, timingSafeEqual } from "node:crypto";

/** Khóa gọi /api/noi-bo/* (pg_cron → pg_net gửi header X-VBP-Noi-Bo). Khóa đặt dưới 32 ký tự coi như chưa cấu hình. */
export function kiemKhoaNoiBo(dau: string | null, khoa: string | undefined): "ok" | "chua_cau_hinh" | "sai" {
  const k = khoa?.trim() ?? "";
  if (k.length < 32) return "chua_cau_hinh";
  if (!dau) return "sai";
  const a = createHash("sha256").update(dau.trim()).digest();
  const b = createHash("sha256").update(k).digest();
  return timingSafeEqual(a, b) ? "ok" : "sai";
}

import type { ThuEmail } from "../email/gui";
import type { PhuThuocKho } from "./phu-thuoc";
import { dinhDangByte, soanThu, type ThuCanGui } from "./thu-nhac";

export const NGUONG_CANH_BAO_BYTE = 200 * 1024 ** 3;

export type PhuThuocDinhKy = PhuThuocKho & {
  guiEmail: (thu: ThuEmail) => Promise<void>;
  baoTelegram: (s: string) => Promise<void>;
};

/** Hàng chờ quá 1 giờ + thùng rác quá 30 ngày: xóa ở CSDL rồi xóa tệp Storage (lỗi Storage chỉ ghi log). */
export async function donKho(pt: PhuThuocKho): Promise<{ tep_cho: number; van_ban: number; loi_xoa: number }> {
  const a = await pt.rpc("kho_don_tep_cho", {});
  const b = await pt.rpc("kho_don_thung_rac", { p_ngay: 30 });
  const duong: string[] = [...(a.duong_dan ?? []), ...(b.duong_dan ?? [])];
  let loiXoa = 0;
  if (duong.length) {
    try {
      await pt.tep.xoa(duong);
    } catch (e) {
      loiXoa = duong.length;
      pt.ghiLog(`[KHO] dọn: xóa ${duong.length} tệp Storage lỗi — ${(e as Error).message}`);
    }
  }
  return { tep_cho: (a.duong_dan ?? []).length, van_ban: Number(b.van_ban ?? 0), loi_xoa: loiXoa };
}

/** Gửi thư nhắc (giữ chỗ trước, gửi hỏng thì bỏ chỗ để lần sau gửi lại) + cảnh báo Telegram khi tổng kho vượt 200 GB. */
export async function nhacEmail(pt: PhuThuocDinhKy): Promise<{ gui: number; loi: number; bo_qua: number }> {
  const r = await pt.rpc("kho_nhac_can_gui", {});
  let gui = 0;
  let loi = 0;
  let boQua = 0;
  for (const t of (r.thu ?? []) as ThuCanGui[]) {
    const g = await pt.rpc("kho_nhac_ghi", { p_loai: t.loai, p_khoa: t.khoa, p_user: t.user_id });
    if (!g.moi) {
      boQua++;
      continue;
    }
    try {
      await pt.guiEmail(soanThu(t));
      gui++;
    } catch (e) {
      loi++;
      await pt.rpc("kho_nhac_bo", { p_loai: t.loai, p_khoa: t.khoa });
      pt.ghiLog(`[KHO] nhắc ${t.loai}: gửi email lỗi — ${(e as Error).message}`);
    }
  }
  const tong = Number(r.tong_byte ?? 0);
  if (tong > NGUONG_CANH_BAO_BYTE) {
    const homNay = new Date().toLocaleDateString("sv-SE", { timeZone: "Asia/Ho_Chi_Minh" });
    const g = await pt.rpc("kho_nhac_ghi", { p_loai: "he_thong_200gb", p_khoa: homNay, p_user: null });
    if (g.moi) {
      await pt.baoTelegram(`⚠️ VanBanPlus: kho tệp đã dùng ${dinhDangByte(tong)}, vượt ngưỡng cảnh báo 200 GB (máy chủ 1 TB theo hợp đồng).`);
    }
  }
  return { gui, loi, bo_qua: boQua };
}

-- Chạy tay một lần trên prod sau migration 011 + sau khi app đã có VBP_NOI_BO_KHOA.
-- Cần trước 2 bí mật Vault (tạo bằng kịch bản đọc tệp khóa, không gõ giá trị vào đây):
--   vbp_noi_bo_url  = gốc API (https://vanbanplus.giakiemso.com)
--   vbp_noi_bo_khoa = đúng giá trị VBP_NOI_BO_KHOA của app
CREATE OR REPLACE FUNCTION kho_goi_noi_bo(p_duong TEXT) RETURNS BIGINT
LANGUAGE plpgsql SECURITY DEFINER SET search_path = public, pg_temp AS $$
DECLARE v_goc TEXT; v_khoa TEXT;
BEGIN
    SELECT decrypted_secret INTO v_goc FROM vault.decrypted_secrets WHERE name = 'vbp_noi_bo_url';
    SELECT decrypted_secret INTO v_khoa FROM vault.decrypted_secrets WHERE name = 'vbp_noi_bo_khoa';
    IF v_goc IS NULL OR v_khoa IS NULL THEN RAISE EXCEPTION 'kho_goi_noi_bo: thiếu bí mật Vault'; END IF;
    RETURN net.http_post(
        url := rtrim(v_goc, '/') || p_duong,
        body := '{}'::jsonb,
        headers := jsonb_build_object('Content-Type', 'application/json', 'X-VBP-Noi-Bo', v_khoa),
        timeout_milliseconds := 60000);
END;
$$;
REVOKE ALL ON FUNCTION kho_goi_noi_bo(TEXT) FROM PUBLIC, anon, authenticated, service_role;

-- Dọn hàng chờ + thùng rác: mỗi giờ phút 37. Nhắc email: 01:13 UTC = 08:13 giờ Việt Nam.
