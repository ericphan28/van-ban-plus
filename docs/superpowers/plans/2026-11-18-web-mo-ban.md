```ts
import { quaNhieu, type KetQua } from "../phien/dich-vu";
import { bat, coNul, coSurrogateLe, sai, type KQ } from "./kiem-dau-vao";
import { tuKiem, xong } from "./loi";
import type { PhuThuocKho } from "./phu-thuoc";

export type CoQuan = { ten_co_quan: string; xa: string; tinh: string; dac_diem: string[]; nguoi_ky: string; chuc_danh: string };
const TRUONG = ["ten_co_quan", "xa", "tinh", "nguoi_ky", "chuc_danh"] as const;
const TOI_DA = 300;
const TOI_DA_BYTE = 8000; // CHECK CSDL 8192 — chặn trước để trả 400 thay vì 500
export const GHI_TOI_DA = 60;
export const GHI_CUA_SO_GIAY = 900;

const xau = (s: string) => s.length > TOI_DA || coNul(s) || coSurrogateLe(s);

export function kiemCoQuan(than: Record<string, unknown>): KQ<CoQuan> {
  return bat(() => {
    const cq: CoQuan = { ten_co_quan: "", xa: "", tinh: "", dac_diem: [], nguoi_ky: "", chuc_danh: "" };
    for (const k of TRUONG) {
      const v = than[k];
      if (v == null) continue;
      if (typeof v !== "string") sai("Thông tin cơ quan không hợp lệ.");
      const s = (v as string).trim();
      if (xau(s)) sai(`Mỗi ô tối đa ${TOI_DA} ký tự.`);
      cq[k] = s;
    }
    if (than.dac_diem != null) {
      if (!Array.isArray(than.dac_diem) || than.dac_diem.length > 10) sai("Đặc điểm tối đa 10 dòng.");
      for (const x of than.dac_diem as unknown[]) {
        if (typeof x !== "string") sai("Đặc điểm không hợp lệ.");
        const s = (x as string).trim();
        if (xau(s)) sai(`Mỗi dòng đặc điểm tối đa ${TOI_DA} ký tự.`);
        if (s) cq.dac_diem.push(s);
      }
    }
    if (Buffer.byteLength(JSON.stringify(cq), "utf8") > TOI_DA_BYTE) sai("Thông tin cơ quan quá dài.");
    return cq;
  });
}

export async function docCoQuan(pt: PhuThuocKho, userId: string): Promise<KetQua> {
  return xong({ co_quan: (await pt.rpc("co_quan_doc", { p_user_id: userId })) ?? null });
}

export async function ghiCoQuan(pt: PhuThuocKho, userId: string, than: Record<string, unknown>): Promise<KetQua> {
  const k = kiemCoQuan(than);
  if ("loi" in k) return tuKiem(k);
  const d = await pt.rpc("tan_suat_dem", { p_khoa: `toi-co-quan:${userId}`, p_cua_so_giay: GHI_CUA_SO_GIAY, p_toi_da: GHI_TOI_DA });
  if (!d?.duoc) return quaNhieu(d?.thu_lai_sau_giay);
  return xong({ co_quan: await pt.rpc("co_quan_ghi", { p_user_id: userId, p_du_lieu: k.du_lieu }) }, "Đã lưu thông tin cơ quan.");
}
```
