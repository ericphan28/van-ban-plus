**Goal:** `vanbanplus-api` cấp phiên đăng nhập cho web/app (JWT 15 phút + token làm mới xoay vòng 30 ngày) qua 4 cách — email + mật khẩu, OTP email, Google, xác thực 2 lớp TOTP — và mọi route v2 nhận `Authorization: Bearer` bên cạnh `X-API-Key`.

**Tech Stack:** Next.js 16 App Router, TypeScript, `node:crypto` (HS256, HMAC, RS256, TOTP — không thêm gói npm), `bcryptjs` (có sẵn), Supabase JS (service role), Postgres 17 (`supabase/postgres:17.6.1.155`), `tsx --test`, `pg` (chỉ bộ kiểm), Gmail API qua `fetch`.

```ts
// src/lib/email/gui.ts
/** Gửi email qua Gmail API (hộp thư công ty). Đổi sang dịch vụ khác: thay taoGuiEmail, giữ chữ ký guiEmail. */

export type ThuEmail = { den: string; tieu_de: string; van_ban: string; html?: string };
export type CauHinhGmail = { client_id: string; client_secret: string; refresh_token: string; tu: string; ten_tu: string };

export function docCauHinhGmail(env: Record<string, string | undefined> = process.env): CauHinhGmail | null {
  const lay = (k: string) => (env[k] ?? "").trim();
  const ch = {
    client_id: lay("VBP_GMAIL_CLIENT_ID"),
    client_secret: lay("VBP_GMAIL_CLIENT_SECRET"),
    refresh_token: lay("VBP_GMAIL_REFRESH_TOKEN"),
    tu: lay("VBP_GMAIL_TU"),
    ten_tu: lay("VBP_GMAIL_TEN_TU") || "VanBanPlus",
  };
  return ch.client_id && ch.client_secret && ch.refresh_token && ch.tu ? ch : null;
}

const maTieuDe = (s: string) => `=?UTF-8?B?${Buffer.from(s, "utf8").toString("base64")}?=`;
const base64Dong = (s: string) => Buffer.from(s, "utf8").toString("base64").replace(/.{76}/g, "$&\r\n");

export function dungMime(thu: ThuEmail, ch: Pick<CauHinhGmail, "tu" | "ten_tu">): string {
  for (const v of [thu.den, thu.tieu_de, ch.tu, ch.ten_tu]) {
    if (/[\r\n]/.test(v)) throw new Error("Địa chỉ hoặc tiêu đề email có ký tự xuống dòng");
  }
  const dau = [
    `From: ${maTieuDe(ch.ten_tu)} <${ch.tu}>`,
    `To: ${thu.den}`,
    `Subject: ${maTieuDe(thu.tieu_de)}`,
    "MIME-Version: 1.0",
  ];
  if (!thu.html) {
    return [...dau, "Content-Type: text/plain; charset=UTF-8", "Content-Transfer-Encoding: base64", "", base64Dong(thu.van_ban)].join("\r\n");
  }
  const ranh = `vbp-${Date.now().toString(36)}-${Math.random().toString(36).slice(2)}`;
  return [
    ...dau,
    `Content-Type: multipart/alternative; boundary="${ranh}"`,
    "",
    `--${ranh}`,
    "Content-Type: text/plain; charset=UTF-8",
    "Content-Transfer-Encoding: base64",
    "",
    base64Dong(thu.van_ban),
    `--${ranh}`,
    "Content-Type: text/html; charset=UTF-8",
    "Content-Transfer-Encoding: base64",
    "",
    base64Dong(thu.html),
    `--${ranh}--`,
  ].join("\r\n");
}

export type PhuThuocEmail = {
  cauHinh: () => CauHinhGmail | null;
  fetch: typeof fetch;
  ghiLog: (s: string) => void;
  bayGio: () => number;
  laProduction: boolean;
};

export function taoGuiEmail(pt: PhuThuocEmail): (thu: ThuEmail) => Promise<void> {
  let token: { gia_tri: string; het: number } | null = null;

  async function layToken(ch: CauHinhGmail): Promise<string> {
    if (token && pt.bayGio() < token.het) return token.gia_tri;
    const r = await pt.fetch("https://oauth2.googleapis.com/token", {
      method: "POST",
      headers: { "Content-Type": "application/x-www-form-urlencoded" },
      body: new URLSearchParams({
        client_id: ch.client_id,
        client_secret: ch.client_secret,
        refresh_token: ch.refresh_token,
        grant_type: "refresh_token",
      }).toString(),
    });
    if (!r.ok) throw new Error(`Gmail token ${r.status}`);
    const j = (await r.json()) as { access_token?: string; expires_in?: number };
    if (!j.access_token) throw new Error("Gmail token rỗng");
    token = { gia_tri: j.access_token, het: pt.bayGio() + Math.max(60, (j.expires_in ?? 3600) - 60) * 1000 };
    return token.gia_tri;
  }

  return async function guiEmail(thu: ThuEmail): Promise<void> {
    const ch = pt.cauHinh();
    if (!ch) {
      pt.ghiLog(
        pt.laProduction
          ? `[EMAIL] chưa cấu hình Gmail — không gửi thư «${thu.tieu_de.replace(/\d{6}/g, "******")}» tới ${thu.den.replace(/^(.).*@/, "$1***@")}`
          : `[EMAIL] chưa cấu hình Gmail — thư tới ${thu.den}: ${thu.tieu_de}\n${thu.van_ban}`
      );
      return;
    }
    const raw = Buffer.from(dungMime(thu, ch), "utf8").toString("base64url");
    const at = await layToken(ch);
    const r = await pt.fetch("https://gmail.googleapis.com/gmail/v1/users/me/messages/send", {
      method: "POST",
      headers: { Authorization: `Bearer ${at}`, "Content-Type": "application/json" },
      body: JSON.stringify({ raw }),
    });
    if (r.status === 401) token = null;
    if (!r.ok) throw new Error(`Gmail ${r.status}`);
  };
}

export const guiEmail = taoGuiEmail({
  cauHinh: () => docCauHinhGmail(),
  fetch: (...a) => globalThis.fetch(...a),
  ghiLog: (s) => console.log(s),
  bayGio: () => Date.now(),
  laProduction: process.env.NODE_ENV === "production",
});
```

```ts
// tests/ho-tro/rpc-pg.ts
import type { Client } from "pg";
import type { GoiRpc } from "../../src/lib/phien/rpc";

/** Gọi hàm SQL trả jsonb bằng tham số đặt tên — cùng cách supabase.rpc gọi trên production. */
export function rpcPg(c: Client): GoiRpc {
  return async (ten, thamSo) => {
    if (!/^[a-z_0-9]+$/.test(ten)) throw new Error(`tên hàm lạ: ${ten}`);
    const khoa = Object.keys(thamSo);
    const sql = `SELECT ${ten}(${khoa.map((k, i) => `${k} => $${i + 1}`).join(", ")}) AS r`;
    const { rows } = await c.query(sql, khoa.map((k) => thamSo[k]));
    return rows[0].r;
  };
}
```

```sql
CREATE OR REPLACE FUNCTION tan_suat_dem(p_khoa TEXT, p_cua_so_giay INT, p_toi_da INT) RETURNS JSONB
LANGUAGE plpgsql SECURITY DEFINER SET search_path = public, pg_temp AS $$
DECLARE v_cua_so TIMESTAMPTZ; v_so INT;
BEGIN
    IF p_cua_so_giay <= 0 OR p_toi_da <= 0 THEN RAISE EXCEPTION 'tan_suat_dem: tham số sai'; END IF;
    v_cua_so := to_timestamp(floor(extract(epoch FROM now()) / p_cua_so_giay) * p_cua_so_giay);
    INSERT INTO dem_tan_suat AS d (khoa, cua_so, so) VALUES (p_khoa, v_cua_so, 1)
        ON CONFLICT (khoa, cua_so) DO UPDATE SET so = d.so + 1
        RETURNING so INTO v_so;
    IF v_so > p_toi_da THEN
        RETURN jsonb_build_object('duoc', false, 'thu_lai_sau_giay',
            GREATEST(1, ceil(extract(epoch FROM v_cua_so + make_interval(secs => p_cua_so_giay) - now())))::int);
    END IF;
    RETURN jsonb_build_object('duoc', true);
END;
$$;
```

```ts
// src/lib/phien/ma.ts
import { createHash, createHmac, randomBytes, randomInt, timingSafeEqual } from "node:crypto";

export const taoTokenNgauNhien = (soByte = 32): string => randomBytes(soByte).toString("base64url");
export const bamHex = (s: string): string => createHash("sha256").update(s, "utf8").digest("hex");
export const hmacHex = (khoa: Buffer, s: string): string => createHmac("sha256", khoa).update(s, "utf8").digest("hex");
export const taoMaSo = (soChu = 6): string => String(randomInt(0, 10 ** soChu)).padStart(soChu, "0");

/** Bỏ i, l, o, 0, 1 để đọc chép không nhầm. */
const CHU_DU_PHONG = "abcdefghjkmnpqrstuvwxyz23456789";

export function taoMaDuPhong(soMa = 10): string[] {
  const ds = new Set<string>();
  while (ds.size < soMa) {
    let s = "";
    for (let i = 0; i < 8; i++) s += CHU_DU_PHONG[randomInt(0, CHU_DU_PHONG.length)];
    ds.add(`${s.slice(0, 4)}-${s.slice(4)}`);
  }
  return [...ds];
}

export function chuanMaDuPhong(s: string): string | null {
  const t = s.toLowerCase().replace(/[\s-]/g, "");
  return /^[a-hjkmnp-z2-9]{8}$/.test(t) ? t : null;
}

export function bangNhau(a: string, b: string): boolean {
  const x = Buffer.from(a, "utf8");
  const y = Buffer.from(b, "utf8");
  return x.length === y.length && timingSafeEqual(x, y);
}
```

```ts
// src/lib/phien/jwt.ts
import { createHmac, timingSafeEqual } from "node:crypto";

export type ClaimPhien = { sub: string; sid: string; iat: number; exp: number };
export type KetQuaJwt = { ok: true; claim: ClaimPhien } | { ok: false; ly_do: "sai" | "het_han" };

/** Phần đầu cố định: kiểm đúng nguyên chuỗi nên không có chuyện đổi «alg». */
const DAU = Buffer.from(JSON.stringify({ alg: "HS256", typ: "JWT" })).toString("base64url");
const SAI: KetQuaJwt = { ok: false, ly_do: "sai" };

export function kyJwt(c: { sub: string; sid: string }, khoa: Buffer, bayGioGiay: number, thoiHanGiay = 900): string {
  const than = Buffer.from(
    JSON.stringify({ sub: c.sub, sid: c.sid, iat: bayGioGiay, exp: bayGioGiay + thoiHanGiay })
  ).toString("base64url");
  const chu = createHmac("sha256", khoa).update(`${DAU}.${than}`).digest("base64url");
  return `${DAU}.${than}.${chu}`;
}

export function kiemJwt(token: string, khoa: Buffer, bayGioGiay: number): KetQuaJwt {
  const phan = token.split(".");
  if (phan.length !== 3 || phan[0] !== DAU) return SAI;
  const mongDoi = createHmac("sha256", khoa).update(`${phan[0]}.${phan[1]}`).digest();
  const nhan = Buffer.from(phan[2], "base64url");
  if (nhan.length !== mongDoi.length || !timingSafeEqual(nhan, mongDoi)) return SAI;
  let c: Partial<ClaimPhien>;
  try {
    c = JSON.parse(Buffer.from(phan[1], "base64url").toString("utf8"));
  } catch {
    return SAI;
  }
  if (typeof c.sub !== "string" || typeof c.sid !== "string" || !Number.isInteger(c.iat) || !Number.isInteger(c.exp)) {
    return SAI;
  }
  if ((c.exp as number) <= bayGioGiay) return { ok: false, ly_do: "het_han" };
```

```ts
// src/lib/phien/totp.ts
import { createHmac } from "node:crypto";
import { bangNhau } from "./ma";

const BANG = "ABCDEFGHIJKLMNOPQRSTUVWXYZ234567";
export const BUOC_GIAY = 30;

export function base32(b: Buffer): string {
  let bit = 0;
  let giaTri = 0;
  let ra = "";
  for (const x of b) {
    giaTri = ((giaTri << 8) | x) & 0xffff;
    bit += 8;
    while (bit >= 5) {
      ra += BANG[(giaTri >>> (bit - 5)) & 31];
      bit -= 5;
    }
  }
  if (bit > 0) ra += BANG[(giaTri << (5 - bit)) & 31];
  return ra;
}

/** RFC 6238 / 4226, HMAC-SHA1. */
export function maTotp(biMat: Buffer, buoc: number, soChu = 6): string {
  const b = Buffer.alloc(8);
  b.writeBigUInt64BE(BigInt(buoc));
  const h = createHmac("sha1", biMat).update(b).digest();
  const o = h[h.length - 1] & 0xf;
  const n = ((h[o] & 0x7f) << 24) | (h[o + 1] << 16) | (h[o + 2] << 8) | h[o + 3];
  return String(n % 10 ** soChu).padStart(soChu, "0");
}

export function timBuocKhop(biMat: Buffer, ma: string, bayGioGiay: number, lech = 1): number | null {
  const t = Math.floor(bayGioGiay / BUOC_GIAY);
  for (let d = -lech; d <= lech; d++) {
    if (bangNhau(maTotp(biMat, t + d), ma)) return t + d;
  }
  return null;
}

export function urlOtpauth(biMat: Buffer, email: string, nha = "VanBanPlus"): string {
  return `otpauth://totp/${encodeURIComponent(`${nha}:${email}`)}?secret=${base32(biMat)}&issuer=${encodeURIComponent(nha)}&algorithm=SHA1&digits=6&period=30`;
}
```

```ts
import { createPublicKey, verify, type JsonWebKey } from "node:crypto";

export type KhoaJwk = { kid: string; kty: string; n: string; e: string; alg?: string; use?: string };
export type ThongTinGoogle = { sub: string; email: string; ho_ten: string };

const NHA_PHAT_HANH = new Set(["accounts.google.com", "https://accounts.google.com"]);

function docPhan(s: string): any {
  return JSON.parse(Buffer.from(s, "base64url").toString("utf8"));
}

export async function kiemIdTokenGoogle(
  token: string,
  tc: { clientId: string; layKhoa: (lamMoi: boolean) => Promise<KhoaJwk[]>; bayGioGiay: number }
): Promise<ThongTinGoogle | null> {
  const p = token.split(".");
  if (p.length !== 3) return null;
  let dau: any;
  let than: any;
  try {
    dau = docPhan(p[0]);
    than = docPhan(p[1]);
  } catch {
    return null;
  }
  if (dau?.alg !== "RS256" || typeof dau.kid !== "string") return null;
  let k = (await tc.layKhoa(false)).find((x) => x.kid === dau.kid);
  if (!k) k = (await tc.layKhoa(true)).find((x) => x.kid === dau.kid);
  if (!k || k.kty !== "RSA") return null;
  let dung = false;
  try {
    const khoa = createPublicKey({ key: { kty: "RSA", n: k.n, e: k.e } as JsonWebKey, format: "jwk" });
    dung = verify("RSA-SHA256", Buffer.from(`${p[0]}.${p[1]}`), khoa, Buffer.from(p[2], "base64url"));
  } catch {
    return null;
  }
  if (!dung) return null;
  if (!NHA_PHAT_HANH.has(than.iss) || than.aud !== tc.clientId) return null;
  if (typeof than.exp !== "number" || than.exp <= tc.bayGioGiay) return null;
  if (typeof than.iat === "number" && than.iat > tc.bayGioGiay + 300) return null;
  if (than.email_verified !== true && than.email_verified !== "true") return null;
  if (typeof than.sub !== "string" || !than.sub || typeof than.email !== "string") return null;
  return { sub: than.sub, email: than.email, ho_ten: typeof than.name === "string" ? than.name.trim().slice(0, 100) : "" };
}

let boNho: { khoa: KhoaJwk[]; het: number; layLuc: number } | null = null;

export async function layKhoaGoogle(lamMoi = false): Promise<KhoaJwk[]> {
  const bay = Date.now();
  if (boNho && bay < boNho.het && !(lamMoi && bay - boNho.layLuc > 60_000)) return boNho.khoa;
  const r = await fetch("https://www.googleapis.com/oauth2/v3/certs");
  if (!r.ok) throw new Error(`Google certs ${r.status}`);
  const j = (await r.json()) as { keys?: KhoaJwk[] };
  const m = /max-age=(\d+)/.exec(r.headers.get("cache-control") ?? "");
  boNho = { khoa: Array.isArray(j.keys) ? j.keys : [], het: bay + (m ? Number(m[1]) : 3600) * 1000, layLuc: bay };
  return boNho.khoa;
}
```

```ts
const MUC_DICH = ["dang-nhap", "dang-ky", "dat-lai-mat-khau"] as const;
export type MucDich = (typeof MUC_DICH)[number];
const docMucDich = (v: unknown): MucDich | null =>
  typeof v === "string" && (MUC_DICH as readonly string[]).includes(v) ? (v as MucDich) : null;

const NHAN_OTP: Record<MucDich, string> = {
  "dang-nhap": "Mã đăng nhập VanBanPlus",
  "dang-ky": "Mã xác nhận đăng ký VanBanPlus",
  "dat-lai-mat-khau": "Mã đặt lại mật khẩu VanBanPlus",
};

const bamOtp = (khoa: Buffer, email: string, mucDich: MucDich, ma: string) => hmacHex(khoa, `otp:${email}:${mucDich}:${ma}`);

export async function guiOtp(pt: PhuThuocPhien, than: Record<string, unknown>, nc: NgonCanh): Promise<KetQua> {
  const khoa = pt.khoaPhien();
  if (!khoa) return LOI.chuaCauHinh();
  const email = chuanEmail(than.email);
  const mucDich = docMucDich(than.muc_dich);
  if (!email || !mucDich) return LOI.dauVaoSai("Email hoặc mục đích không hợp lệ.");
  const ip = await pt.rpc<{ duoc: boolean; thu_lai_sau_giay?: number }>("tan_suat_dem", {
    p_khoa: `otp-ip:${nc.ip}`, p_cua_so_giay: CUA_SO, p_toi_da: 10,
  });
  if (!ip.duoc) return quaNhieu(ip.thu_lai_sau_giay);
  const ma = taoMaSo(6);
  const r = await pt.rpc<{ duoc: boolean; thu_lai_sau_giay?: number }>("otp_tao", {
    p_email: email, p_muc_dich: mucDich, p_bam_hex: bamOtp(khoa, email, mucDich, ma),
  });
  if (!r.duoc) return quaNhieu(r.thu_lai_sau_giay);
  try {
    await pt.guiEmail({
      den: email,
      tieu_de: `${NHAN_OTP[mucDich]}: ${ma}`,
      van_ban: `${NHAN_OTP[mucDich]}: ${ma}\n\nMã có hiệu lực 10 phút. Không chia sẻ mã này cho người khác.\n`
        + "Không yêu cầu mã này thì bỏ qua email.\n\nVanBanPlus — Gia Kiệm Số",
    });
  } catch (e) {
    console.error("[PHIEN] gửi OTP lỗi", (e as Error).message);
    return loi(502, "EMAIL_LOI", "Không gửi được email. Thử lại sau.");
  }
  return { loai: "xong", message: "Đã gửi mã xác nhận tới email (nếu email hợp lệ). Mã có hiệu lực 10 phút." };
}

export async function xacNhanOtp(pt: PhuThuocPhien, than: Record<string, unknown>, nc: NgonCanh): Promise<KetQua> {
  const khoa = pt.khoaPhien();
  if (!khoa) return LOI.chuaCauHinh();
  const email = chuanEmail(than.email);
  const mucDich = docMucDich(than.muc_dich);
  const ma = typeof than.ma === "string" ? than.ma.trim() : "";
  if (!email || !mucDich) return LOI.dauVaoSai("Email hoặc mục đích không hợp lệ.");
  if (!/^\d{6}$/.test(ma)) return LOI.otpSai();

  let matKhau: string | null = null;
  let hoTen: string | null = null;
  if (mucDich !== "dang-nhap") {
    const l = kiemMatKhau(than.mat_khau);
    if (l) return LOI.dauVaoSai(l);
    matKhau = than.mat_khau as string;
  }
  if (mucDich === "dang-ky") {
    hoTen = chuanHoTen(than.ho_ten);
    if (!hoTen) return LOI.dauVaoSai("Họ tên không hợp lệ.");
  }

  const ip = await pt.rpc<{ duoc: boolean; thu_lai_sau_giay?: number }>("tan_suat_dem", {
    p_khoa: `otp-kiem-ip:${nc.ip}`, p_cua_so_giay: CUA_SO, p_toi_da: 30,
  });
  if (!ip.duoc) return quaNhieu(ip.thu_lai_sau_giay);

  const r = await pt.rpc<{ kq: "ok" | "sai" | "het_han" }>("otp_kiem", {
    p_email: email, p_muc_dich: mucDich, p_bam_hex: bamOtp(khoa, email, mucDich, ma),
  });
  if (r.kq === "sai") return LOI.otpSai();
  if (r.kq !== "ok") return LOI.otpHetHan();

  let user = await pt.rpc<ApiUser | null>("nguoi_dung_theo_email", { p_email: email });

  if (mucDich === "dang-ky") {
    if (user) return loi(409, "EMAIL_DA_CO", "Email đã có tài khoản. Đăng nhập hoặc đặt lại mật khẩu.");
    const moi = await taoNguoiDung(pt, { email, hoTen: hoTen!, matKhau, xacMinh: true });
    if (!moi) return loi(409, "EMAIL_DA_CO", "Email đã có tài khoản. Đăng nhập hoặc đặt lại mật khẩu.");
    await ghiNhatKy(pt, moi.id, "dang_ky", nc, { cach: "otp" });
    return capPhien(pt, moi.id, "otp", nc);
  }

  if (mucDich === "dang-nhap" && !user) {
    user = (await taoNguoiDung(pt, { email, hoTen: email.split("@")[0], matKhau: null, xacMinh: true }))
      ?? (await pt.rpc<ApiUser | null>("nguoi_dung_theo_email", { p_email: email }));
    if (user) await ghiNhatKy(pt, user.id, "dang_ky", nc, { cach: "otp" });
  }
  if (!user) return loi(404, "KHONG_TIM_THAY", "Không có tài khoản với email này.");
  if (user.is_active === false) return LOI.taiKhoanNgung();
  await pt.rpc("nguoi_dung_xac_minh_email", { p_user: user.id });

  if (mucDich === "dat-lai-mat-khau") {
    await pt.rpc("nguoi_dung_dat_mat_khau", { p_user: user.id, p_bam: await pt.bamMatKhau(matKhau!) });
    await pt.rpc("phien_thu_hoi", { p_user: user.id, p_phien: null, p_tru: null, p_ly_do: "dat_lai_mat_khau" });
    await ghiNhatKy(pt, user.id, "dat_lai_mat_khau", nc);
  } else {
    await pt.rpc("dang_nhap_ghi_dung", { p_user: user.id, p_ip: nc.ip });
  }
  return hoanTatDangNhap(pt, user, "otp", nc);
}
```

```ts
function biMatCua(khoa: Buffer, userId: string, d: DongHaiLop): Buffer {
  return giaiMa(Buffer.from(d.bi_mat_hex, "hex"), Buffer.from(d.iv_hex, "hex"), khoa, ganTotp(userId));

/** Mã 6 số → TOTP (không cho dùng lại bước); dạng xxxx-xxxx → mã dự phòng một lần. */
export async function kiemMaHaiLop(pt: PhuThuocPhien, userId: string, ma: string): Promise<boolean | "chua_cau_hinh"> {
  const khoa = pt.khoaTotp();
  if (!khoa) return "chua_cau_hinh";
  const d = await pt.rpc<DongHaiLop | null>("hai_lop_doc", { p_user: userId });
  if (!d || !d.bat) return false;
  const chu = ma.replace(/\s/g, "");
  if (/^\d{6}$/.test(chu)) {
    const buoc = timBuocKhop(biMatCua(khoa, userId, d), chu, giay(pt));
    if (buoc === null) return false;
    return (await pt.rpc<{ duoc: boolean }>("hai_lop_dung_buoc", { p_user: userId, p_buoc: buoc })).duoc;
  }
  const dp = chuanMaDuPhong(chu);
  if (!dp) return false;
  return (await pt.rpc<{ duoc: boolean }>("hai_lop_dung_du_phong", {
    p_user: userId, p_bam_hex: bamDuPhong(khoa, userId, dp),
  })).duoc;
}
```

```ts
import { NextResponse, type NextRequest } from "next/server";
import type { ApiUser } from "@/types";
import { xacThucChiTiet } from "../tac-vu/xac-thuc";
import { cookieDat, cookieXoa, TEN_COOKIE } from "./cookie";
import { loi, LOI, type KetQua, type NgonCanh } from "./dich-vu";
import { tomTatThietBi } from "./thiet-bi";

const KHONG_LUU = { "Cache-Control": "no-store" };

export function traLoi(kq: KetQua): NextResponse {
  if (kq.loai === "phien") {
    const r = NextResponse.json(
      { success: true, message: "Đã đăng nhập.", data: { access_token: kq.access_token, het_han_giay: kq.het_han_giay, nguoi_dung: kq.nguoi_dung } },
      { headers: KHONG_LUU }
    );
    r.headers.append("Set-Cookie", cookieDat(kq.token_lam_moi));
    return r;
  }
  if (kq.loai === "can_2_lop") {
    return NextResponse.json(
      { success: true, message: "Nhập mã xác thực 2 lớp.", code: "CAN_2_LOP", data: { can_2_lop: true, ma_tam: kq.ma_tam } },
      { headers: KHONG_LUU }
    );
  }
  if (kq.loai === "xong") {
    return NextResponse.json({ success: true, message: kq.message, data: kq.data ?? null }, { headers: KHONG_LUU });
  }
  const r = NextResponse.json(
    { success: false, message: kq.message, code: kq.code, ...(kq.data ? { data: kq.data } : {}) },
    { status: kq.http, headers: KHONG_LUU }
  );
  if (kq.code === "PHIEN_THU_HOI") r.headers.append("Set-Cookie", cookieXoa());
  return r;
}

/** Sau Cloudflare: cf-connecting-ip là IP thật; x-forwarded-for đầu tiên khách tự đặt được nên để cuối. */
export function ipKhach(req: NextRequest): string {
  return (
    req.headers.get("cf-connecting-ip")?.trim() ||
    req.headers.get("x-real-ip")?.trim() ||
    req.headers.get("x-forwarded-for")?.split(",")[0]?.trim() ||
    "khong-ro"
  ).slice(0, 64);
}

export const ngonCanh = (req: NextRequest): NgonCanh => ({
  ip: ipKhach(req),
  thiet_bi: tomTatThietBi(req.headers.get("user-agent")),
});

export async function docThan(req: NextRequest): Promise<Record<string, unknown>> {
  const t = await req.json().catch(() => null);
  return t && typeof t === "object" && !Array.isArray(t) ? (t as Record<string, unknown>) : {};
}

export function tokenLamMoi(req: NextRequest, than: Record<string, unknown>): string | null {
  return req.cookies.get(TEN_COOKIE)?.value || (typeof than.token_lam_moi === "string" ? than.token_lam_moi : null);
}

const LOI_XAC_THUC = {
  CHUA_DANG_NHAP: LOI.chuaDangNhap,
  PHIEN_HET_HAN: LOI.phienHetHan,
  PHIEN_THU_HOI: LOI.phienThuHoi,
} as const;

/** Route tài khoản / thiết bị: chỉ nhận phiên web / app (Bearer), không nhận X-API-Key. */
export async function canPhien(req: NextRequest): Promise<{ user: ApiUser; sid: string } | NextResponse> {
  const r = await xacThucChiTiet(req);
  if ("loi" in r) return traLoi(LOI_XAC_THUC[r.loi]());
  if (!r.sid) return traLoi(loi(401, "CHUA_DANG_NHAP", "Cần đăng nhập bằng tài khoản web."));
  return { user: r.user, sid: r.sid };
}

export async function chay(fn: () => Promise<NextResponse>): Promise<NextResponse> {
  try {
    return await fn();
  } catch (e) {
    console.error("[PHIEN] lỗi bất ngờ", (e as Error).message);
    return traLoi(loi(500, "LOI_MAY_CHU", "Lỗi máy chủ. Thử lại sau."));
  }
}
```

```ts
import { NextResponse, type NextRequest } from "next/server";
import { docNguon, quyetDinhCors } from "./lib/phien/cors";

/**
 * Proxy — CORS cho mọi /api (quyết định ở src/lib/phien/cors.ts).
 * v2 + nguồn trong VBP_CORS_NGUON: trả đúng nguồn + credentials (web / app dùng cookie làm mới).
 * Route phiên / tài khoản: nguồn lạ không ghi được (403), không đọc được. Route khác giữ «*» cho bản cài máy.
 */
export function proxy(request: NextRequest) {
  const qd = quyetDinhCors(
    request.nextUrl.pathname,
    request.method,
    request.headers.get("origin"),
    docNguon(process.env.VBP_CORS_NGUON)
  );
  if (qd.chan) {
    return NextResponse.json(
      { success: false, message: "Nguồn gọi không được phép.", code: "NGUON_LA" },
      { status: 403, headers: qd.dau }
    );
  }
  if (request.method === "OPTIONS") return new NextResponse(null, { status: 204, headers: qd.dau });
  const response = NextResponse.next();
  for (const [k, v] of Object.entries(qd.dau)) response.headers.set(k, v);
  return response;
}

export const config = {
  matcher: "/api/:path*",
};

export default proxy;
```
