# L5 bản web VanBanPlus — mẫu văn bản + tra cứu pháp quy (API + web): kế hoạch thực hiện

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Đưa lên production phân hệ «Mẫu văn bản» (80 mẫu có sẵn chỉ đọc + mẫu riêng của từng tài khoản) và «Tra cứu pháp quy» (NĐ 30/2020, 86 mục, tìm không dấu) — API trong `vanbanplus-api`, giao diện trong `vanbanplus-web`.

**Architecture:** Migration `012_mau_phap_quy.sql` tạo `mau_chung` + `phap_quy` (dữ liệu chung, chỉ đọc) và `kho_mau` (mẫu riêng, có chủ sở hữu như mọi bảng kho). Dữ liệu chung lấy từ bản cài máy: công cụ C# (`tools/xuat-du-lieu-web`, repo gốc) xuất JSON từ `AIVanBan.Core` → tệp sinh `supabase/du-lieu/012_mau_phap_quy_du_lieu.sql` gọi hàm nạp chạy lại được. Mọi luật nằm trong hàm SQL `SECURITY DEFINER` trả `jsonb`, gọi qua `GoiRpc`; lớp dịch vụ `src/lib/mau/*`, `src/lib/phap-quy/*` trả `KetQua`; route mỏng dùng `canNguoiDung` + `chay` của L3. Web: hai phân hệ `src/phan-he/mau/`, `src/phan-he/phap-quy/`, gọi API qua `goi()` của `src/api/khach.ts`, CSS riêng từng phân hệ (nạp cùng chunk lười).

**Tech Stack:** API: Next.js 16 App Router, TypeScript, Postgres 17 (`unaccent`, `tsvector` — dùng `kho_khong_dau` của 010), `node:test` qua `tsx`. Công cụ xuất: .NET 9 (tham chiếu `AIVanBan.Core`). Web: React 18 + Vite 6 + React Router 7 + os-kit, Vitest + Testing Library, Playwright (chụp).

**Spec:** `docs/superpowers/specs/2026-11-17-ban-web-dot-a-design.md` (mục 5, 7, 8) · hợp đồng `docs/superpowers/plans/2026-11-17-hop-dong-web-dot-a.md` (mục 1, 2, 3, 5, 6, 7, 9). Đọc cả hai trước Task 1.

## Hợp đồng API L5 (Task 0 ghi vào hợp đồng chung §5b để L4 / L6 thấy sớm)

```ts
type TruongMau = { ma: string; nhan: string };            // ma: ^[A-Za-z][A-Za-z0-9_]{0,63}$ — khóa `truong` của tác vụ soan-thao
type MauTomTat = { nguon: "chung" | "rieng"; id: string;   // chung: mã chữ (vd san-cong-van) · rieng: uuid
  ten: string; loai: LoaiVanBan; nhom: string | null; mo_ta: string | null; the: string[]; co_noi_dung: boolean;
  moi: boolean; pho_bien: boolean; tu_mau_chung: string | null; tao_luc: string; sua_luc: string; phien_ban: number };
type Mau = MauTomTat & { noi_dung_mau: string | null; truong: TruongMau[] };
type NutCay = { ma: string; cha: string | null; loai: "van_ban" | "chuong" | "muc" | "dieu" | "phu_luc" | "phan"; tieu_de: string; thu_tu: number };
type MucPhapQuy = NutCay & { noi_dung: string; the: string[]; duong_dan: { ma: string; tieu_de: string }[];
  con: { ma: string; loai: string; tieu_de: string }[]; truoc: { ma: string; tieu_de: string } | null; sau: { ma: string; tieu_de: string } | null };
```

| Route | Thân / tham số | Trả |
| --- | --- | --- |
| `GET /api/v2/mau` | `q` (không dấu), `nguon` (`chung`\|`rieng`), `loai`, `nhom`, `trang`, `so_dong` ≤ 100 (mặc định 50) | `{ muc: MauTomTat[], tong, nhom: [{ ten, so }] }` — mẫu riêng lên trước (mới sửa trước), rồi mẫu có sẵn theo thứ tự bản cài máy; có `q` thì xếp theo độ khớp |
| `GET /api/v2/mau/chung/[ma]` | — | `Mau` · 404 |
| `POST /api/v2/mau/rieng` | `{ ten, loai, nhom?, mo_ta?, noi_dung_mau?, truong?: [{ nhan, ma? }], the? }` hoặc `{ tu_mau_chung: ma, ten? }` | `Mau` · 403 `HET_DUNG_LUONG` · 400 · 404 · 429 |
| `GET /api/v2/mau/rieng/[id]` | — | `Mau` · 404 |
| `PATCH /api/v2/mau/rieng/[id]` | `{ phien_ban, ...trường cần đổi }` | `Mau` · 409 `XUNG_DOT_PHIEN_BAN` (`data.ban_moi`) · 400 · 404 · 429 |
| `DELETE /api/v2/mau/rieng/[id]` | — | `null` · 404 · 429 (xóa hẳn, mẫu không có tệp) |
| `GET /api/v2/phap-quy` | — | `NutCay[]` (cả cây, không kèm nội dung) |
| `GET /api/v2/phap-quy/[ma]` | `ma` dạng `nd-30-2020` hoặc `nd-30-2020:dieu-8` | `MucPhapQuy` · 404 |
| `GET /api/v2/phap-quy/tim` | `q` | `{ muc: [{ ma, loai, tieu_de, duong_dan: string, trich: string }], tong }` (≤ 50 mục) |

// Dữ liệu chung của bản web (mẫu có sẵn + pháp quy) lấy từ bản cài máy: JSON (tools/xuat-du-lieu-web ở repo gốc)
// + public/template-store.json → dòng bảng → tệp SQL nạp (supabase/du-lieu/012_mau_phap_quy_du_lieu.sql).
import { readFileSync } from "node:fs";
import { join } from "node:path";
import { LOAI_VAN_BAN, nhanTruong, type LoaiVanBan } from "../tac-vu/soan-thao/prompt";

const KHOA_TRUONG = /^[A-Za-z][A-Za-z0-9_]{0,63}$/;
const MA_MAU = /^[a-z0-9-]{1,80}$/;
const MA_PHAP_QUY = /^[a-z0-9-]{1,60}(:[A-Za-z0-9-]{1,60})?$/;
export const THE_DOLLAR = "$vbp_du_lieu$";

export type TruongMau = { ma: string; nhan: string };
export type MauCaiSan = {
  ham: string; ten: string; loai: string; nhom: string; mo_ta: string; noi_dung_mau: string; goi_y_ai: string;
  truong: string[]; the: string[];
};
export type MauCuaHang = {
  store_id: string; name: string; type: string; category: string; description: string; template_content: string;
  ai_prompt: string; required_fields: string[]; tags: string[]; is_new?: boolean; is_popular?: boolean;
};
export type DongMauChung = {
  ma: string; nguon: "cai_san" | "cua_hang"; thu_tu: number; ten: string; loai: LoaiVanBan; nhom: string | null;
  mo_ta: string | null; noi_dung_mau: string | null; goi_y_ai: string | null; truong: TruongMau[]; the: string[];
  moi: boolean; pho_bien: boolean;
};
export type NutNguon = { ma: string; cha: string | null; loai: string; thu_tu: number; tieu_de: string; noi_dung: string; the: string[] };
export type LoaiNut = "van_ban" | "chuong" | "muc" | "dieu" | "phu_luc" | "phan";
export type DongPhapQuy = { ma: string; cha: string | null; loai: LoaiNut; thu_tu: number; tieu_de: string; noi_dung: string; the: string[] };

/** «CreateUBNDBaoCaoKTXHTemplate» → «san-ubnd-bao-cao-ktxh». Mã ổn định theo tên hàm, không theo tên hiển thị. */
export function maTuHam(ham: string): string {
  const goc = ham.replace(/^Create/, "").replace(/Template$/, "");
  return "san-" + goc.replace(/([a-z0-9])([A-Z])/g, "$1-$2").replace(/([A-Z])([A-Z][a-z])/g, "$1-$2").toLowerCase();
}

const rong = (s: string | null | undefined): string | null => {
  const t = (s ?? "").replace(/\r\n/g, "\n").trim();
  return t ? t : null;
};
const sachThe = (ds: string[] | null | undefined) => [...new Set((ds ?? []).map((t) => t.trim()).filter(Boolean))];

/** Nhãn trường bỏ biểu tượng đầu (nhanTruong của tác vụ soạn thảo có «🏢 », «📨 »…). */
const nhan = (ma: string) => nhanTruong(ma).replace(/^[^\p{L}\p{N}]+/u, "").trim();

function dsTruong(ds: string[] | null | undefined, ngu: string): TruongMau[] {
  const ra: TruongMau[] = [];
  for (const ma of ds ?? []) {
    if (!KHOA_TRUONG.test(ma)) throw new Error(`${ngu}: khóa trường lạ «${ma}»`);
    if (!ra.some((t) => t.ma === ma)) ra.push({ ma, nhan: nhan(ma) });
  }
  return ra;
}

function loai(v: string, ngu: string): LoaiVanBan {
  if (!(LOAI_VAN_BAN as readonly string[]).includes(v)) throw new Error(`${ngu}: loại văn bản lạ «${v}»`);
  return v as LoaiVanBan;
}

export function dungMauChung(caiSan: MauCaiSan[], cuaHang: MauCuaHang[]): DongMauChung[] {
  const ra: DongMauChung[] = [
    ...caiSan.map((m, i): DongMauChung => ({
      ma: maTuHam(m.ham), nguon: "cai_san", thu_tu: i, ten: m.ten.trim(), loai: loai(m.loai, m.ham),
      nhom: rong(m.nhom), mo_ta: rong(m.mo_ta), noi_dung_mau: rong(m.noi_dung_mau), goi_y_ai: rong(m.goi_y_ai),
      truong: dsTruong(m.truong, m.ham), the: sachThe(m.the), moi: false, pho_bien: false,
    })),
    ...cuaHang.map((m, i): DongMauChung => ({
      ma: m.store_id, nguon: "cua_hang", thu_tu: 1000 + i, ten: m.name.trim(), loai: loai(m.type, m.store_id),
      nhom: rong(m.category), mo_ta: rong(m.description), noi_dung_mau: rong(m.template_content), goi_y_ai: rong(m.ai_prompt),
      truong: dsTruong(m.required_fields, m.store_id), the: sachThe(m.tags), moi: m.is_new === true, pho_bien: m.is_popular === true,
    })),
  ];
  const daCo = new Set<string>();
  for (const d of ra) {
    if (!MA_MAU.test(d.ma)) throw new Error(`mã mẫu sai dạng: ${d.ma}`);
    if (!d.ten) throw new Error(`mẫu thiếu tên: ${d.ma}`);
    if (daCo.has(d.ma)) throw new Error(`mã mẫu trùng: ${d.ma}`);
    daCo.add(d.ma);
  }
  return ra;
}

const LOAI_NUT: Record<string, LoaiNut> = {
  Document: "van_ban", Chapter: "chuong", Section: "muc", Article: "dieu", Appendix: "phu_luc", SubSection: "phan",
};

export function dungPhapQuy(nut: NutNguon[]): DongPhapQuy[] {
  const ra = nut.map((n): DongPhapQuy => {
    const l = LOAI_NUT[n.loai];
    if (!l) throw new Error(`${n.ma}: loại mục lạ «${n.loai}»`);
    if (!MA_PHAP_QUY.test(n.ma)) throw new Error(`mã pháp quy sai dạng: ${n.ma}`);
    return { ma: n.ma, cha: n.cha, loai: l, thu_tu: n.thu_tu, tieu_de: n.tieu_de.trim(), noi_dung: (n.noi_dung ?? "").replace(/\r\n/g, "\n"), the: sachThe(n.the) };
  });
  const ma = new Set(ra.map((p) => p.ma));
  if (ma.size !== ra.length) throw new Error("mã pháp quy trùng");
  for (const p of ra) if (p.cha && !ma.has(p.cha)) throw new Error(`${p.ma}: thiếu mục cha ${p.cha}`);
  return ra;
}

export function sinhSql(mau: DongMauChung[], pq: DongPhapQuy[]): string {
  const khoi = (ds: unknown[]) => {
    const j = "[\n" + ds.map((d) => JSON.stringify(d)).join(",\n") + "\n]";
    if (j.includes(THE_DOLLAR)) throw new Error(`dữ liệu chứa ${THE_DOLLAR}`);
    return `${THE_DOLLAR}${j}${THE_DOLLAR}::jsonb`;
  };
  return [
    "-- Sinh bằng: npx tsx scripts/sinh-du-lieu-mau-phap-quy.ts — KHÔNG sửa tay.",
    "-- Nạp mẫu chung + pháp quy (cần 012_mau_phap_quy.sql). Chạy lại được: thêm / sửa / bỏ dòng cho khớp nguồn.",
    `SELECT mau_chung_nap(${khoi(mau)});`,
    `SELECT phap_quy_nap(${khoi(pq)});`,
    "",
  ].join("\n");
}

const docJson = (tep: string) => JSON.parse(readFileSync(tep, "utf8"));

-- ---------- 2. trợ giúp (nội bộ) ----------
-- Từ khóa đã bỏ dấu (tachTuKhoa ở API) → tsquery tiền tố, mọi từ phải có. Từ sai dạng bị bỏ (chống chèn).
CREATE OR REPLACE FUNCTION mau_tsquery(p_tu_khoa JSONB) RETURNS TSQUERY
LANGUAGE sql IMMUTABLE SET search_path = pg_catalog, pg_temp AS $$
    SELECT to_tsquery('simple', string_agg(quote_literal(t) || ':*', ' & '))
      FROM jsonb_array_elements_text(CASE WHEN jsonb_typeof(p_tu_khoa) = 'array' THEN p_tu_khoa ELSE '[]'::jsonb END) AS t
     WHERE t ~ '^[a-z0-9]{1,40}$';
$$;

-- Trước khi tạo mẫu riêng: khóa theo chủ (lệnh tạo song song không vượt trần), kho chỉ đọc → HET_DUNG_LUONG, trần 500.
-- NULL = được tạo. Gọi bên trong giao dịch của hàm tạo (khóa nhả khi giao dịch xong).
CREATE OR REPLACE FUNCTION kho_mau_kiem_tao(p_chu_loai TEXT, p_chu_id UUID) RETURNS JSONB
LANGUAGE plpgsql SET search_path = public, pg_temp AS $$
BEGIN
    PERFORM kho_kiem_chu(p_chu_loai, p_chu_id);
    PERFORM pg_advisory_xact_lock(hashtextextended('kho_mau:' || p_chu_loai || ':' || p_chu_id::TEXT, 0));
    IF (kho_dung_luong_xem(p_chu_loai, p_chu_id)->>'chi_doc')::BOOLEAN THEN
        RETURN jsonb_build_object('loi', 'HET_DUNG_LUONG');
    END IF;
    IF (SELECT count(*) FROM kho_mau WHERE chu_loai = p_chu_loai AND chu_id = p_chu_id) >= 500 THEN
        RETURN kho_loi_dau_vao('Tối đa 500 mẫu riêng. Xóa bớt mẫu không dùng rồi tạo lại.');
    END IF;
    RETURN NULL;
END;
$$;

-- ---------- 3. mẫu: tìm, đọc, ghi ----------
-- p_loc: { tu_khoa: text[] (đã bỏ dấu), nguon? 'chung'|'rieng', loai?, nhom?, trang?, so_dong? (≤ 100, mặc định 50) }.
-- Không từ khóa: mẫu riêng trước (mới sửa trước), rồi mẫu chung theo thu_tu. Có từ khóa: xếp theo độ khớp trước.
CREATE OR REPLACE FUNCTION mau_tim(p_chu_loai TEXT, p_chu_id UUID, p_loc JSONB) RETURNS JSONB
LANGUAGE plpgsql STABLE SECURITY DEFINER SET search_path = public, pg_temp AS $$
DECLARE
    v_q TSQUERY := mau_tsquery(p_loc->'tu_khoa');
    v_nguon TEXT := p_loc->>'nguon';
    v_so INT; v_trang INT; v_tong BIGINT; v_muc JSONB; v_nhom JSONB;
BEGIN
    PERFORM kho_kiem_chu(p_chu_loai, p_chu_id);
    v_so := LEAST(GREATEST(COALESCE((p_loc->>'so_dong')::INT, 50), 1), 100);
    v_trang := LEAST(GREATEST(COALESCE((p_loc->>'trang')::INT, 1), 1), 10000);

    WITH loc AS (
        SELECT 'rieng'::TEXT AS nguon, k.id::TEXT AS id, 0 AS uu_tien, NULL::INT AS thu_tu, k.sua_luc,
               CASE WHEN v_q IS NULL THEN 0 ELSE ts_rank(k.tim, v_q) END AS hang
          FROM kho_mau k
         WHERE k.chu_loai = p_chu_loai AND k.chu_id = p_chu_id
           AND (v_nguon IS NULL OR v_nguon = 'rieng')
           AND (v_q IS NULL OR k.tim @@ v_q)
           AND (p_loc->>'loai' IS NULL OR k.loai = p_loc->>'loai')
           AND (p_loc->>'nhom' IS NULL OR k.nhom = p_loc->>'nhom')
        UNION ALL
        SELECT 'chung', m.ma, 1, m.thu_tu, m.sua_luc,
               CASE WHEN v_q IS NULL THEN 0 ELSE ts_rank(m.tim, v_q) END
          FROM mau_chung m
         WHERE (v_nguon IS NULL OR v_nguon = 'chung')
           AND (v_q IS NULL OR m.tim @@ v_q)
           AND (p_loc->>'loai' IS NULL OR m.loai = p_loc->>'loai')
           AND (p_loc->>'nhom' IS NULL OR m.nhom = p_loc->>'nhom')
    ), xep AS (
        SELECT nguon, id, row_number() OVER (ORDER BY hang DESC, uu_tien,
                                             CASE WHEN uu_tien = 0 THEN sua_luc END DESC NULLS LAST,
                                             thu_tu NULLS LAST, id) AS rn
          FROM loc
    )
    SELECT (SELECT count(*) FROM loc),
           COALESCE((SELECT jsonb_agg(CASE WHEN x.nguon = 'rieng' THEN kho_mau_json(k, FALSE) ELSE mau_chung_json(m, FALSE) END
                                      ORDER BY x.rn)
                       FROM xep x
                       LEFT JOIN kho_mau k ON x.nguon = 'rieng' AND k.id::TEXT = x.id
                       LEFT JOIN mau_chung m ON x.nguon = 'chung' AND m.ma = x.id
                      WHERE x.rn > (v_trang - 1) * v_so AND x.rn <= v_trang * v_so), '[]'::jsonb)
      INTO v_tong, v_muc;

    SELECT COALESCE(jsonb_agg(jsonb_build_object('ten', n.nhom, 'so', n.so) ORDER BY n.so DESC, n.nhom), '[]'::jsonb)
      INTO v_nhom
      FROM (SELECT nhom, count(*) AS so
              FROM (SELECT nhom FROM mau_chung WHERE nhom IS NOT NULL
                    UNION ALL
                    SELECT nhom FROM kho_mau WHERE chu_loai = p_chu_loai AND chu_id = p_chu_id AND nhom IS NOT NULL) x
             GROUP BY nhom) n;

    RETURN jsonb_build_object('muc', v_muc, 'tong', v_tong, 'nhom', v_nhom);
END;
$$;

CREATE OR REPLACE FUNCTION mau_chung_doc(p_ma TEXT) RETURNS JSONB
LANGUAGE sql STABLE SECURITY DEFINER SET search_path = public, pg_temp AS $$
    SELECT mau_chung_json(m, TRUE) FROM mau_chung m WHERE m.ma = p_ma;
$$;

-- p_du_lieu (đã kiểm ở API): { ten, loai, nhom?, mo_ta?, noi_dung_mau?, truong: [{ma, nhan}], the: [] }
CREATE OR REPLACE FUNCTION kho_mau_tao(p_chu_loai TEXT, p_chu_id UUID, p_tao_boi UUID, p_du_lieu JSONB) RETURNS JSONB
LANGUAGE plpgsql SECURITY DEFINER SET search_path = public, pg_temp AS $$
DECLARE v_loi JSONB; k kho_mau%ROWTYPE;
BEGIN
    v_loi := kho_mau_kiem_tao(p_chu_loai, p_chu_id);
    IF v_loi IS NOT NULL THEN RETURN v_loi; END IF;
    INSERT INTO kho_mau (chu_loai, chu_id, ten, loai, nhom, mo_ta, noi_dung_mau, truong, the, tao_boi)
    VALUES (p_chu_loai, p_chu_id, p_du_lieu->>'ten', p_du_lieu->>'loai', p_du_lieu->>'nhom', p_du_lieu->>'mo_ta',
            p_du_lieu->>'noi_dung_mau',
            CASE WHEN jsonb_typeof(p_du_lieu->'truong') = 'array' THEN p_du_lieu->'truong' ELSE '[]'::jsonb END,
            kho_mang(p_du_lieu->'the'), p_tao_boi)
    RETURNING * INTO k;
    RETURN jsonb_build_object('ok', TRUE, 'mau', kho_mau_json(k, TRUE));
END;
$$;

-- Chép một mẫu chung thành mẫu riêng (nội dung lấy trong CSDL, máy khách không gửi). p_ten NULL = giữ tên mẫu chung.
CREATE OR REPLACE FUNCTION kho_mau_tu_chung(p_chu_loai TEXT, p_chu_id UUID, p_tao_boi UUID, p_ma TEXT, p_ten TEXT) RETURNS JSONB
LANGUAGE plpgsql SECURITY DEFINER SET search_path = public, pg_temp AS $$
DECLARE v_loi JSONB; m mau_chung%ROWTYPE; k kho_mau%ROWTYPE;
BEGIN
    PERFORM kho_kiem_chu(p_chu_loai, p_chu_id);
    SELECT * INTO m FROM mau_chung WHERE ma = p_ma;
    IF NOT FOUND THEN RETURN jsonb_build_object('loi', 'KHONG_TIM_THAY'); END IF;
    v_loi := kho_mau_kiem_tao(p_chu_loai, p_chu_id);
    IF v_loi IS NOT NULL THEN RETURN v_loi; END IF;
    INSERT INTO kho_mau (chu_loai, chu_id, ten, loai, nhom, mo_ta, noi_dung_mau, truong, the, tu_mau_chung, tao_boi)
    VALUES (p_chu_loai, p_chu_id, COALESCE(p_ten, m.ten), m.loai, m.nhom, m.mo_ta, m.noi_dung_mau, m.truong, m.the, m.ma,
            p_tao_boi)
    RETURNING * INTO k;
    RETURN jsonb_build_object('ok', TRUE, 'mau', kho_mau_json(k, TRUE));
END;
$$;

CREATE OR REPLACE FUNCTION kho_mau_doc(p_chu_loai TEXT, p_chu_id UUID, p_id UUID) RETURNS JSONB

-- p_du_lieu chỉ chứa trường cần đổi (khóa có mặt = đổi, kể cả null). Không kiểm chỉ đọc (spec §6: vẫn sửa được).
CREATE OR REPLACE FUNCTION kho_mau_sua(p_chu_loai TEXT, p_chu_id UUID, p_id UUID, p_phien_ban INT, p_du_lieu JSONB) RETURNS JSONB
LANGUAGE plpgsql SECURITY DEFINER SET search_path = public, pg_temp AS $$
DECLARE k kho_mau%ROWTYPE;
BEGIN
    PERFORM kho_kiem_chu(p_chu_loai, p_chu_id);
    SELECT * INTO k FROM kho_mau WHERE id = p_id AND chu_loai = p_chu_loai AND chu_id = p_chu_id FOR UPDATE;
    IF NOT FOUND THEN RETURN jsonb_build_object('loi', 'KHONG_TIM_THAY'); END IF;
    IF k.phien_ban <> p_phien_ban THEN
        RETURN jsonb_build_object('loi', 'XUNG_DOT_PHIEN_BAN', 'ban_moi', kho_mau_json(k, TRUE));
    END IF;
    UPDATE kho_mau
       SET ten = CASE WHEN p_du_lieu ? 'ten' THEN p_du_lieu->>'ten' ELSE ten END,
           loai = CASE WHEN p_du_lieu ? 'loai' THEN p_du_lieu->>'loai' ELSE loai END,
           nhom = CASE WHEN p_du_lieu ? 'nhom' THEN p_du_lieu->>'nhom' ELSE nhom END,
           mo_ta = CASE WHEN p_du_lieu ? 'mo_ta' THEN p_du_lieu->>'mo_ta' ELSE mo_ta END,
           noi_dung_mau = CASE WHEN p_du_lieu ? 'noi_dung_mau' THEN p_du_lieu->>'noi_dung_mau' ELSE noi_dung_mau END,
           truong = CASE WHEN jsonb_typeof(p_du_lieu->'truong') = 'array' THEN p_du_lieu->'truong' ELSE truong END,
           the = CASE WHEN p_du_lieu ? 'the' THEN kho_mang(p_du_lieu->'the') ELSE the END,
           sua_luc = now(), phien_ban = phien_ban + 1
     WHERE id = k.id
    RETURNING * INTO k;
    RETURN jsonb_build_object('ok', TRUE, 'mau', kho_mau_json(k, TRUE));
END;
$$;

-- ---------- 4. pháp quy ----------
CREATE OR REPLACE FUNCTION phap_quy_cay() RETURNS JSONB
LANGUAGE sql STABLE SECURITY DEFINER SET search_path = public, pg_temp AS $$
    SELECT COALESCE(jsonb_agg(jsonb_build_object('ma', ma, 'cha', cha, 'loai', loai, 'tieu_de', tieu_de, 'thu_tu', thu_tu)
                              ORDER BY cha NULLS FIRST, thu_tu, ma), '[]'::jsonb)
      FROM phap_quy;
$$;

-- Một mục + đường dẫn từ gốc + mục con + mục liền trước / liền sau theo thứ tự đọc (duyệt sâu, cùng văn bản).
CREATE OR REPLACE FUNCTION phap_quy_doc(p_ma TEXT) RETURNS JSONB
LANGUAGE plpgsql STABLE SECURITY DEFINER SET search_path = public, pg_temp AS $$
DECLARE p phap_quy%ROWTYPE; v_duong JSONB; v_con JSONB; v_truoc JSONB; v_sau JSONB;
BEGIN
    SELECT * INTO p FROM phap_quy WHERE ma = p_ma;
    IF NOT FOUND THEN RETURN NULL; END IF;

    WITH RECURSIVE len AS (
        SELECT q.ma, q.cha, q.tieu_de, 1 AS bac FROM phap_quy q WHERE q.ma = p.cha
        UNION ALL
        SELECT q.ma, q.cha, q.tieu_de, len.bac + 1 FROM phap_quy q JOIN len ON q.ma = len.cha WHERE len.bac < 20
    )
    SELECT COALESCE(jsonb_agg(jsonb_build_object('ma', ma, 'tieu_de', tieu_de) ORDER BY bac DESC), '[]'::jsonb)
      INTO v_duong FROM len;

    SELECT COALESCE(jsonb_agg(jsonb_build_object('ma', ma, 'loai', loai, 'tieu_de', tieu_de) ORDER BY thu_tu, ma), '[]'::jsonb)
      INTO v_con FROM phap_quy WHERE cha = p.ma;

    WITH RECURSIVE cay AS (
        SELECT q.ma, q.tieu_de, ARRAY[q.thu_tu] AS duong
          FROM phap_quy q WHERE q.ma = split_part(p.ma, ':', 1) AND q.cha IS NULL
        UNION ALL
        SELECT q.ma, q.tieu_de, cay.duong || q.thu_tu
          FROM phap_quy q JOIN cay ON q.cha = cay.ma WHERE cardinality(cay.duong) < 20
    ), xep AS (
        SELECT ma, tieu_de, row_number() OVER (ORDER BY duong, ma) AS rn FROM cay
    ), vi_tri AS (
        SELECT rn FROM xep WHERE ma = p.ma
    )
    SELECT (SELECT jsonb_build_object('ma', x.ma, 'tieu_de', x.tieu_de) FROM xep x, vi_tri v WHERE x.rn = v.rn - 1),
           (SELECT jsonb_build_object('ma', x.ma, 'tieu_de', x.tieu_de) FROM xep x, vi_tri v WHERE x.rn = v.rn + 1)
      INTO v_truoc, v_sau;

    RETURN jsonb_build_object('ma', p.ma, 'cha', p.cha, 'loai', p.loai, 'tieu_de', p.tieu_de, 'thu_tu', p.thu_tu,
                              'noi_dung', p.noi_dung, 'the', to_jsonb(p.the), 'duong_dan', v_duong, 'con', v_con,
                              'truoc', v_truoc, 'sau', v_sau);
END;
$$;

-- p_loc: { tu_khoa: text[] (đã bỏ dấu), gioi_han? (≤ 100, mặc định 50) }. Trả kèm noi_dung để API cắt đoạn trích.
CREATE OR REPLACE FUNCTION phap_quy_tim(p_loc JSONB) RETURNS JSONB
LANGUAGE plpgsql STABLE SECURITY DEFINER SET search_path = public, pg_temp AS $$
DECLARE
    v_q TSQUERY := mau_tsquery(p_loc->'tu_khoa');
    v_gh INT := LEAST(GREATEST(COALESCE((p_loc->>'gioi_han')::INT, 50), 1), 100);
    v_tong BIGINT; v_muc JSONB;
BEGIN
    IF v_q IS NULL THEN RETURN jsonb_build_object('muc', '[]'::jsonb, 'tong', 0); END IF;
    SELECT count(*) INTO v_tong FROM phap_quy WHERE tim @@ v_q;
    SELECT COALESCE(jsonb_agg(jsonb_build_object('ma', x.ma, 'cha', x.cha, 'loai', x.loai, 'tieu_de', x.tieu_de,
                                                 'noi_dung', x.noi_dung) ORDER BY x.hang DESC, x.cha NULLS FIRST, x.thu_tu), '[]'::jsonb)
      INTO v_muc
      FROM (SELECT q.ma, q.cha, q.loai, q.tieu_de, q.noi_dung, q.thu_tu, ts_rank(q.tim, v_q) AS hang
              FROM phap_quy q WHERE q.tim @@ v_q
             ORDER BY hang DESC, q.cha NULLS FIRST, q.thu_tu LIMIT v_gh) x;
    RETURN jsonb_build_object('muc', v_muc, 'tong', v_tong);
END;
$$;

-- ---------- 5. nạp dữ liệu chung (chỉ psql quyền chủ CSDL; tệp supabase/du-lieu/012_mau_phap_quy_du_lieu.sql) ----------
-- Thêm / sửa (chỉ dòng có đổi: sua_luc, phien_ban + 1) / bỏ dòng không còn trong nguồn. Trả { them, sua, bo }.
CREATE OR REPLACE FUNCTION mau_chung_nap(p JSONB) RETURNS JSONB
LANGUAGE plpgsql SET search_path = public, pg_temp AS $$
DECLARE v_them BIGINT; v_sua BIGINT; v_bo BIGINT;
BEGIN
    IF jsonb_typeof(p) IS DISTINCT FROM 'array' OR jsonb_array_length(p) = 0 THEN
        RAISE EXCEPTION 'mau_chung_nap: dữ liệu phải là mảng khác rỗng';
    END IF;
    WITH nguon AS (
        SELECT x.ma, x.nguon, x.thu_tu, x.ten, x.loai, NULLIF(x.nhom, '') AS nhom, NULLIF(x.mo_ta, '') AS mo_ta,
               NULLIF(x.noi_dung_mau, '') AS noi_dung_mau, NULLIF(x.goi_y_ai, '') AS goi_y_ai,
               COALESCE(x.truong, '[]'::jsonb) AS truong, kho_mang(x.the) AS the,
               COALESCE(x.moi, FALSE) AS moi, COALESCE(x.pho_bien, FALSE) AS pho_bien
          FROM jsonb_to_recordset(p) AS x(ma TEXT, nguon TEXT, thu_tu INT, ten TEXT, loai TEXT, nhom TEXT, mo_ta TEXT,
                                         noi_dung_mau TEXT, goi_y_ai TEXT, truong JSONB, the JSONB, moi BOOLEAN,
                                         pho_bien BOOLEAN)
    ), bo AS (
        DELETE FROM mau_chung m WHERE NOT EXISTS (SELECT 1 FROM nguon n WHERE n.ma = m.ma) RETURNING 1
    ), ghi AS (
        INSERT INTO mau_chung AS m (ma, nguon, thu_tu, ten, loai, nhom, mo_ta, noi_dung_mau, goi_y_ai, truong, the, moi, pho_bien)
        SELECT ma, nguon, thu_tu, ten, loai, nhom, mo_ta, noi_dung_mau, goi_y_ai, truong, the, moi, pho_bien FROM nguon
        ON CONFLICT (ma) DO UPDATE
           SET nguon = EXCLUDED.nguon, thu_tu = EXCLUDED.thu_tu, ten = EXCLUDED.ten, loai = EXCLUDED.loai,
               nhom = EXCLUDED.nhom, mo_ta = EXCLUDED.mo_ta, noi_dung_mau = EXCLUDED.noi_dung_mau,
               goi_y_ai = EXCLUDED.goi_y_ai, truong = EXCLUDED.truong, the = EXCLUDED.the, moi = EXCLUDED.moi,
               pho_bien = EXCLUDED.pho_bien, sua_luc = now(), phien_ban = m.phien_ban + 1
         WHERE (m.nguon, m.thu_tu, m.ten, m.loai, m.nhom, m.mo_ta, m.noi_dung_mau, m.goi_y_ai, m.truong, m.the, m.moi, m.pho_bien)
               IS DISTINCT FROM
               (EXCLUDED.nguon, EXCLUDED.thu_tu, EXCLUDED.ten, EXCLUDED.loai, EXCLUDED.nhom, EXCLUDED.mo_ta,
                EXCLUDED.noi_dung_mau, EXCLUDED.goi_y_ai, EXCLUDED.truong, EXCLUDED.the, EXCLUDED.moi, EXCLUDED.pho_bien)
        RETURNING (xmax = 0) AS la_moi
    )
    SELECT (SELECT count(*) FROM bo), count(*) FILTER (WHERE la_moi), count(*) FILTER (WHERE NOT la_moi)
      INTO v_bo, v_them, v_sua FROM ghi;
    RETURN jsonb_build_object('them', v_them, 'sua', v_sua, 'bo', v_bo);
END;
$$;

CREATE OR REPLACE FUNCTION phap_quy_nap(p JSONB) RETURNS JSONB
LANGUAGE plpgsql SET search_path = public, pg_temp AS $$
DECLARE v_them BIGINT; v_sua BIGINT; v_bo BIGINT;
BEGIN
    IF jsonb_typeof(p) IS DISTINCT FROM 'array' OR jsonb_array_length(p) = 0 THEN
        RAISE EXCEPTION 'phap_quy_nap: dữ liệu phải là mảng khác rỗng';
    END IF;
    WITH nguon AS (
        SELECT x.ma, x.cha, x.loai, x.thu_tu, x.tieu_de, COALESCE(x.noi_dung, '') AS noi_dung, kho_mang(x.the) AS the
          FROM jsonb_to_recordset(p) AS x(ma TEXT, cha TEXT, loai TEXT, thu_tu INT, tieu_de TEXT, noi_dung TEXT, the JSONB)
    ), bo AS (
        DELETE FROM phap_quy q WHERE NOT EXISTS (SELECT 1 FROM nguon n WHERE n.ma = q.ma) RETURNING 1
    ), ghi AS (
        INSERT INTO phap_quy AS q (ma, cha, loai, thu_tu, tieu_de, noi_dung, the)
        SELECT ma, cha, loai, thu_tu, tieu_de, noi_dung, the FROM nguon
        ON CONFLICT (ma) DO UPDATE
           SET cha = EXCLUDED.cha, loai = EXCLUDED.loai, thu_tu = EXCLUDED.thu_tu, tieu_de = EXCLUDED.tieu_de,
               noi_dung = EXCLUDED.noi_dung, the = EXCLUDED.the, sua_luc = now(), phien_ban = q.phien_ban + 1
         WHERE (q.cha, q.loai, q.thu_tu, q.tieu_de, q.noi_dung, q.the)
               IS DISTINCT FROM (EXCLUDED.cha, EXCLUDED.loai, EXCLUDED.thu_tu, EXCLUDED.tieu_de, EXCLUDED.noi_dung, EXCLUDED.the)
        RETURNING (xmax = 0) AS la_moi
    )
    SELECT (SELECT count(*) FROM bo), count(*) FILTER (WHERE la_moi), count(*) FILTER (WHERE NOT la_moi)
      INTO v_bo, v_them, v_sua FROM ghi;
    RETURN jsonb_build_object('them', v_them, 'sua', v_sua, 'bo', v_bo);
END;
$$;

/** «Cơ quan nhận» → «co_quan_nhan». Không bắt đầu bằng chữ → thêm «truong_». */
export function maTruong(nhan: string): string {
  const s = boDau(nhan).replace(/[^a-z0-9]+/g, "_").replace(/^_+|_+$/g, "").slice(0, 56);
  if (/^[a-z]/.test(s)) return s;
  return s ? `truong_${s}` : "truong";
}

function kiemTruong(v: unknown): TruongMau[] {
  if (v === null || v === undefined) return [];
  if (!Array.isArray(v)) sai("Trường cần điền phải là danh sách.");
  const ds = v as unknown[];
  if (ds.length > TRUONG_TOI_DA) sai(`Tối đa ${TRUONG_TOI_DA} trường cần điền.`);
  const ra: TruongMau[] = [];
  const daCo = new Set<string>();
  for (const x of ds) {
    if (!x || typeof x !== "object" || Array.isArray(x)) sai("Trường cần điền không hợp lệ.");
    const o = x as Record<string, unknown>;
    const nhan = chuoiTuy(o.nhan, "Tên trường", 200);
    if (!nhan) sai("Trường cần điền thiếu tên.");
    let ma: string;

/** Thân JSON có trần: route Next không giới hạn kích thước thân, đọc luồng và dừng khi vượt (chống làm đầy bộ nhớ). */
export const THAN_TOI_DA_BYTE = 1_048_576;

/** null = quá trần (route trả 400 DAU_VAO_SAI). JSON hỏng / không phải đối tượng → {} (bước kiểm báo thiếu trường). */
export async function docThanCoTran(req: Request, toiDa = THAN_TOI_DA_BYTE): Promise<Record<string, unknown> | null> {
  const khai = Number(req.headers.get("content-length") ?? "0");
  if (Number.isFinite(khai) && khai > toiDa) return null;
  if (!req.body) return {};
  const doc = req.body.getReader();
  const manh: Uint8Array[] = [];
  let tong = 0;
  for (;;) {
    const { done, value } = await doc.read();
    if (done) break;
    tong += value.byteLength;
    if (tong > toiDa) {
      await doc.cancel().catch(() => undefined);
      return null;
    }
    manh.push(value);
  }
  try {
    const t = JSON.parse(Buffer.concat(manh).toString("utf8"));
    return t && typeof t === "object" && !Array.isArray(t) ? (t as Record<string, unknown>) : {};
  } catch {
    return {};
  }
}

/** Tham số đường dẫn có thể còn mã hóa (%3A). Giải mã hỏng → giữ nguyên để bước kiểm dạng chặn. */
export function giaiMaThamSo(s: string): string {
  if (!s.includes("%")) return s;
  try {
    return decodeURIComponent(s);
  } catch {
    return s;
  }
}

import { boDau } from "../kho/kiem-dau-vao";

export type NutCay = { ma: string; cha: string | null; loai: string; tieu_de: string; thu_tu: number };

const gon = (s: string) => s.replace(/\s+/g, " ").trim();
const catNgan = (s: string, n: number) => {
  const g = gon(s);
  return g.length > n ? g.slice(0, n).trimEnd() + "…" : g;
};

/** Vị trí đầu tiên từ khóa t (a-z0-9, từ tachTuKhoa) khớp ĐẦU một từ trong chuỗi đã bỏ dấu; -1 nếu không có. */
function timDauTu(khongDau: string, t: string): number {
  const m = new RegExp(`(^|[^a-z0-9])${t}`).exec(khongDau);
  return m ? m.index + m[1].length : -1;
}

/**
 * Đoạn trích quanh từ khóa khớp sớm nhất (như ExtractMatchContext của bản cài máy). Chữ tiếng Việt dạng NFC bỏ dấu
 * giữ nguyên độ dài nên vị trí trên chuỗi bỏ dấu dùng thẳng cho chuỗi gốc; độ dài lệch (ký tự lạ) → trả phần đầu.
 */
export function trichDoan(noiDung: string, tuKhoa: string[], truoc = 40, sau = 120): string {
  const goc = noiDung.normalize("NFC");
  const khong = boDau(goc);
  if (khong.length !== goc.length) return catNgan(goc, truoc + sau);
  let vt = -1;
  let dai = 0;
  for (const t of tuKhoa) {
    const i = timDauTu(khong, t);
    if (i >= 0 && (vt < 0 || i < vt)) { vt = i; dai = t.length; }
  }
  if (vt < 0) return catNgan(goc, truoc + sau);
  const dau = Math.max(0, vt - truoc);
  const cuoi = Math.min(goc.length, vt + dai + sau);
  return (dau > 0 ? "…" : "") + gon(goc.slice(dau, cuoi)) + (cuoi < goc.length ? "…" : "");
}

/** «Nghị định … › Chương II» — tên các mục từ gốc tới `cha` (gồm cả `cha`). */
export function duongDan(cha: string | null, bang: Map<string, NutCay>): string {
  const ds: string[] = [];
  let c = cha;
  for (let i = 0; c && i < 20; i++) {
    const n = bang.get(c);
    if (!n) break;
    ds.unshift(n.tieu_de);
    c = n.cha;
  }
  return ds.join(" › ");
}

import { thamSoChu, type Chu } from "../kho/chu-so-huu";

import type { PhuThuocKho } from "../kho/phu-thuoc";
import { docLocMau, kiemMau, laMaMauChung } from "./kiem-dau-vao";

/** Ghi mẫu riêng (tạo / sửa / xóa): 120 lần / 15 phút / tài khoản (tan_suat_dem của 009). */
export const GHI_TOI_DA = 120;
export const GHI_CUA_SO_GIAY = 900;

/** Kết quả SQL { ok, <khoa> } | { loi } → KetQua; xung đột phiên bản đổi câu báo cho đúng «mẫu». */
function tuSqlMau(r: any, khoa: string | null, message = "Thành công"): KetQua {
  if (r && typeof r === "object" && "loi" in r) {
    if (r.loi === "XUNG_DOT_PHIEN_BAN") {
      return loi(409, "XUNG_DOT_PHIEN_BAN", "Mẫu vừa được sửa ở nơi khác. Tải lại bản mới rồi sửa tiếp.", { ban_moi: r.ban_moi });
    }
    return tuLoiSql(r);
  }
  return xong(khoa ? r[khoa] : null, message);
}

async function demGhi(pt: PhuThuocKho, user: { id: string }): Promise<KetQua | null> {
  const d = await pt.rpc("tan_suat_dem", { p_khoa: `mau-ghi:${user.id}`, p_cua_so_giay: GHI_CUA_SO_GIAY, p_toi_da: GHI_TOI_DA });
  if (d?.duoc) return null;
  return loi(429, "QUA_NHIEU_YEU_CAU", "Thao tác quá nhiều lần. Thử lại sau ít phút.", { thu_lai_sau_giay: d?.thu_lai_sau_giay ?? 60 });
}

export async function timMau(pt: PhuThuocKho, chu: Chu, sp: URLSearchParams): Promise<KetQua> {
  const l = docLocMau(sp);
  if ("loi" in l) return tuKiem(l);
  return xong(await pt.rpc("mau_tim", { ...thamSoChu(chu), p_loc: l.du_lieu }));
}

export async function docMauChung(pt: PhuThuocKho, ma: string): Promise<KetQua> {
  if (!laMaMauChung(ma)) return LOI_KHO.khongTimThay();
  const m = await pt.rpc("mau_chung_doc", { p_ma: ma });
  return m ? xong(m) : LOI_KHO.khongTimThay();
}

/** Tạo mẫu riêng, hoặc chép mẫu chung khi thân có `tu_mau_chung`. Kho chỉ đọc → ném LoiHetDungLuong (hợp đồng §5). */
export async function taoMauRieng(pt: PhuThuocKho, user: { id: string }, chu: Chu, than: Record<string, unknown>): Promise<KetQua> {
  const q = await demGhi(pt, user);
  if (q) return q;
  if (Object.hasOwn(than, "tu_mau_chung")) {
    if (!laMaMauChung(than.tu_mau_chung)) return LOI_KHO.khongTimThay();
    let ten: string | null = null;
    if (Object.hasOwn(than, "ten")) {
      const k = kiemMau({ ten: than.ten }, "sua");
      if ("loi" in k) return tuKiem(k);
      ten = k.du_lieu.ten ?? null;
    }
    await kiemConGhi(chu, pt.rpc);
    const r = await pt.rpc("kho_mau_tu_chung", { ...thamSoChu(chu), p_tao_boi: user.id, p_ma: than.tu_mau_chung, p_ten: ten });
    return tuSqlMau(r, "mau", "Đã lưu thành mẫu riêng.");
  }
  const k = kiemMau(than, "tao");
  if ("loi" in k) return tuKiem(k);
  await kiemConGhi(chu, pt.rpc);
  const r = await pt.rpc("kho_mau_tao", { ...thamSoChu(chu), p_tao_boi: user.id, p_du_lieu: k.du_lieu });

/** Giá trị trễ `ms` (ô tìm: chỉ gọi API khi ngừng gõ). */
export function useTre<T>(giaTri: T, ms = 300): T {
  const [v, setV] = useState(giaTri);
  useEffect(() => {
    const t = setTimeout(() => setV(giaTri), ms);
    return () => clearTimeout(t);
  }, [giaTri, ms]);
  return v;
}

export function thanGui(n: Nhap, cu: Mau | null): DuLieuMau {
  const maCu = new Map((cu?.truong ?? []).map((t) => [t.nhan, t.ma]));

      .map((nhan) => (maCu.has(nhan) ? { nhan, ma: maCu.get(nhan)! } : { nhan })),

// Hàm thuần của phân hệ pháp quy: tách bảng trong nội dung, tô sáng từ khóa không dấu, xếp mục lục, liên kết tính năng.
import type { NutCay } from "./api";

export type Khoi = { loai: "doan"; dong: string[] } | { loai: "bang"; dau: string[]; than: string[][] };

/** Nội dung là chữ thuần; bảng viết kiểu Markdown («| a | b |» + dòng «|---|»). */
export function tachKhoi(noiDung: string): Khoi[] {
  const ra: Khoi[] = [];
  let doan: string[] = [];
  let bang: string[][] = [];
  const dayDoan = () => { if (doan.length) { ra.push({ loai: "doan", dong: doan }); doan = []; } };
  const dayBang = () => {
    if (bang.length) { const [dau, ...than] = bang; ra.push({ loai: "bang", dau, than }); bang = []; }
  };
  for (const dong of noiDung.replace(/\r\n/g, "\n").split("\n")) {
    const t = dong.trim();
    if (t.length > 1 && t.startsWith("|") && t.endsWith("|")) {
      dayDoan();
      const o = t.slice(1, -1).split("|").map((x) => x.trim());
      if (o.every((x) => /^:?-{2,}:?$/.test(x))) continue;
      bang.push(o);
    } else if (t === "") {
      dayBang(); dayDoan();
    } else {
      dayBang(); doan.push(dong);
    }
  }
  dayBang(); dayDoan();
  return ra;
}

/** Chữ thường, bỏ dấu (cả đ) — giống boDau của API. */
export function boDau(s: string): string {
  return s.normalize("NFD").replace(/\p{M}+/gu, "").replace(/đ/g, "d").replace(/Đ/g, "D").toLowerCase();
}

/** Tách chữ thành đoạn thường / đoạn tô sáng: từ khóa (bỏ dấu) khớp đầu từ. Độ dài lệch sau bỏ dấu → không tô. */
export function toSang(chu: string, q: string): { chu: string; sang: boolean }[] {
  const goc = chu.normalize("NFC");
  const khong = boDau(goc);
  const tu = [...new Set(boDau(q).split(/[^a-z0-9]+/).filter(Boolean))].slice(0, 8);
  if (!tu.length || !goc || khong.length !== goc.length) return [{ chu: goc, sang: false }];
  const dau = new Array<boolean>(goc.length).fill(false);
  for (const t of tu) {
    const re = new RegExp(`(^|[^a-z0-9])(${t})`, "g");
    let m: RegExpExecArray | null;
    while ((m = re.exec(khong))) {
      const i = m.index + m[1].length;
      for (let k = i; k < i + t.length; k++) dau[k] = true;
      re.lastIndex = i + t.length;
    }
  }
  const ra: { chu: string; sang: boolean }[] = [];
  for (let i = 0; i < goc.length; ) {
    let j = i;
    while (j < goc.length && dau[j] === dau[i]) j++;
    ra.push({ chu: goc.slice(i, j), sang: dau[i] });
    i = j;
  }
  return ra;
}

/** Mục lục theo thứ tự đọc (duyệt sâu, con theo thu_tu) + bậc thụt lề. */
export function xepCay(ds: NutCay[]): { nut: NutCay; bac: number }[] {
  const con = new Map<string | null, NutCay[]>();
  for (const n of ds) {
    const k = n.cha;
    if (!con.has(k)) con.set(k, []);
    con.get(k)!.push(n);
  }
  for (const v of con.values()) v.sort((a, b) => a.thu_tu - b.thu_tu || a.ma.localeCompare(b.ma));
  const ra: { nut: NutCay; bac: number }[] = [];
  const daQua = new Set<string>();
  const di = (cha: string | null, bac: number) => {
    if (bac > 20) return;
    for (const n of con.get(cha) ?? []) {
      if (daQua.has(n.ma)) continue;
      daQua.add(n.ma);
      ra.push({ nut: n, bac });
      di(n.ma, bac + 1);
    }
  };
  di(null, 0);
  return ra;
}

const khop = (chu, q) => boDau(q).split(/[^a-z0-9]+/).filter(Boolean).every((t) => new RegExp(`(^|[^a-z0-9])${t}`).test(boDau(chu)));

function docMuc(ma) {
  const p = PQ.find((x) => x.ma === ma);
  if (!p) return null;
  const duong = [];
  for (let c = p.cha; c; c = PQ.find((x) => x.ma === c)?.cha) duong.unshift({ ma: c, tieu_de: PQ.find((x) => x.ma === c).tieu_de });
  const thuTu = [];
  const di = (cha) => PQ.filter((x) => x.cha === cha).sort((a, b) => a.thu_tu - b.thu_tu).forEach((x) => { thuTu.push(x); di(x.ma); });
  di(null);
  const i = thuTu.indexOf(p);
  const lk = (x) => (x ? { ma: x.ma, tieu_de: x.tieu_de } : null);
  return { ...p, duong_dan: duong, con: PQ.filter((x) => x.cha === ma).map(({ ma: m, loai, tieu_de }) => ({ ma: m, loai, tieu_de })),
    truoc: lk(thuTu[i - 1]), sau: lk(thuTu[i + 1]) };
}

/** r = đường sau /api/v2 (đã bỏ tiền tố). Không phải route L5 → null. */
export function traLoiL5(phuongThuc, r, sp) {
  if (r === "/mau" && phuongThuc === "GET") {
    const q = sp.get("q") ?? "";
    const ds = MAU.filter((m) => (!sp.get("nguon") || m.nguon === sp.get("nguon")) && (!sp.get("nhom") || m.nhom === sp.get("nhom"))
      && (!q || khop(`${m.ten} ${m.mo_ta ?? ""} ${m.nhom ?? ""}`, q)));
    return ok({ muc: ds.map(tom), tong: ds.length + (q || sp.get("nguon") === "rieng" ? 0 : 74), nhom: NHOM });
  }
  const mc = /^\/mau\/(chung|rieng)\/([^/]+)$/.exec(r);
  if (mc && phuongThuc === "GET") {
    const m = MAU.find((x) => x.nguon === mc[1] && x.id === decodeURIComponent(mc[2]));
    return m ? ok(m) : loi(404, "KHONG_TIM_THAY", "Không tìm thấy.");
  }
  if (r === "/phap-quy" && phuongThuc === "GET") return ok(PQ.map(({ noi_dung, the, ...x }) => x));
  if (r === "/phap-quy/tim") {
    const q = sp.get("q") ?? "";
    const ds = PQ.filter((p) => khop(`${p.tieu_de} ${p.noi_dung}`, q));
    return ok({ tong: ds.length, muc: ds.map((p) => ({ ma: p.ma, loai: p.loai, tieu_de: p.tieu_de,
      duong_dan: docMuc(p.ma).duong_dan.map((d) => d.tieu_de).join(" › "),
      trich: p.noi_dung.replace(/\s+/g, " ").slice(0, 160) + (p.noi_dung.length > 160 ? "…" : "") })) });
  }
  const pq = /^\/phap-quy\/([^/]+)$/.exec(r);
  if (pq) {
    const m = docMuc(decodeURIComponent(pq[1]));
    return m ? ok(m) : loi(404, "KHONG_TIM_THAY", "Không tìm thấy.");
  }
  return null;
}
